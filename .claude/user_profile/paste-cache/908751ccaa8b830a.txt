    /**
     * inferDominantContext - Uses vocabulary frequency + regex rules to determine
     * the primary interaction context for an entity based on its available actions.
     */
    private inferDominantContext(actions: string[]): string {
        const contextScores = new Map<string, number>();
        
        for (const action of actions) {
            const freq = getActionFrequency(action);
            for (const rule of CONTEXT_RULES) {
                if (rule.pattern.test(action)) {
                    const weightedScore = freq * rule.priority;
                    contextScores.set(rule.context, (contextScores.get(rule.context) || 0) + weightedScore);
                }
            }
        }

        let bestContext = 'ground';
        let bestScore = 0;
        
        for (const [ctx, score] of contextScores) {
            if (score > bestScore) {
                bestScore = score;
                bestContext = ctx;
            }
        }

        return bestContext;
    }

    /**
     * Updated mapToResolved - Injects actionFrequencies, authentic classification,
     * and dominant context inference.
     */
    private mapToResolved(raw: any, category: ResolvedEntity['category'], truth: ManifestEntity | undefined, fallbackId: number): ResolvedEntity {
        const sovereignId = truth?.id || `S-${category.toUpperCase()}-${fallbackId}`;
        const actions = (truth?.actions || raw.actions || []);
        
        // Build action frequency map from vocabulary corpus
        const actionFrequencies = new Map<string, number>();
        for (const action of actions) {
            actionFrequencies.set(action, getActionFrequency(action));
        }

        // Authentic classification: manifest presence OR anchor registry match
        const isAuthentic = !!truth || AUTHENTIC_ANCHORS.has(truth?.title || raw.name || '');

        // Layering: if truth has sub-entities, treat as layers
        const layers = truth?.subEntities || [];

        return {
            id: fallbackId,
            canonicalName: truth?.title || raw.name || `Entity ${fallbackId}`,
            displayName: truth?.title || raw.name || `Entity ${fallbackId}`,
            examine: truth?.meta?.examine || raw.examine || 'Observation inconclusive.',
            category,
            actions,
            actionFrequencies,
            isAuthentic,
            confidence: truth ? (isAuthentic ? 1.0 : 0.95) : 0.7,
            groundingSource: truth ? 'manifest' : 'cache',
            variants: [], // Reserved for future variant mapping
            layers,
            modelRef: { id: sovereignId }
        };
    }
