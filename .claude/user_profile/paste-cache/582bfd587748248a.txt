```typescript
// D:\POG2\src\services\SovereignEntityResolver.ts

import { ModelRolodex } from '../api/ai/ModelRolodex';
import { CacheForensicsLimb } from '../limbs/CacheForensicsLimb';

// ── Types ──────────────────────────────────────────────────────────

export interface ManifestEntry {
  id: number;
  name?: string;
  modelId?: number;
  modelName?: string;
  examine?: string;
  actions?: string[];
  category?: string;
  version?: number;
  isNote?: boolean;
  isBank?: boolean;
  isWield?: boolean;
  // Raw manifest may have contiguous unnamed entries
  _rawIndex: number;
}

export interface ResolvedEntity {
  id: number;
  canonicalName: string;
  displayName: string;
  examine: string;
  category: 'npc' | 'item' | 'object' | 'ground_item' | 'bank_note';
  modelRef: { id: number; name: string; version: number };
  actions: string[];
  variants: VariantSpec[];
  isAuthentic: boolean;
  layers: ManifestEntry[]; // Collapsed contiguous entries
  confidence: number; // 0.0-1.0
  groundingSource: 'manifest' | 'cache' | 'inferred';
}

interface VariantSpec {
  context: 'inventory' | 'bank' | 'shop' | 'trade' | 'wield' | 'ground';
  modelId?: number;
  actions: string[];
  isNote: boolean;
}

// ── Authentic Anchor Registry ──────────────────────────────────────

const AUTHENTIC_ANCHORS = new Set([
  'Banker', 'Bank booth', 'Guard', 'General Store assistant',
  'Shop assistant', 'Shop keeper', 'Grand Exchange clerk',
  'Fairy ring', 'Spirit tree', 'Lodestone', 'Portal',
  'Stairs', 'Ladder', 'Door', 'Gate'
]);

// ── Interaction Vocabulary ─────────────────────────────────────────

const INTERACTION_MAP: Record<string, VariantSpec['context']> = {
  'Wield': 'wield',
  'Wear': 'wield',
  'Equip': 'wield',
  'Withdraw': 'bank',
  'Deposit': 'bank',
  'Bank': 'bank',
  'Buy': 'shop',
  'Sell': 'shop',
  'Trade': 'trade',
  'Exchange': 'trade',
  'Pick up': 'ground',
  'Take': 'ground',
  'Examine': 'ground',
  'Use': 'inventory',
  'Drop': 'inventory',
  'Eat': 'inventory',
  'Drink': 'inventory',
};

// ── Sovereign Shorthand: Layering Engine ───────────────────────────

function collapseLayers(rawEntries: ManifestEntry[]): ManifestEntry[][] {
  const groups: ManifestEntry[][] = [];
  let currentGroup: ManifestEntry[] = [];

  for (const entry of rawEntries) {
    if (entry.name && currentGroup.length > 0) {
      // Named entry closes previous group, starts new one
      groups.push(currentGroup);
      currentGroup = [entry];
    } else if (!entry.name && currentGroup.length > 0) {
      // Unnamed entry appends to current group as layer
      currentGroup.push(entry);
    } else if (entry.name) {
      // Named entry, no open group
      currentGroup = [entry];
    }
    // Edge: unnamed entry with no open group → discard or orphan handling
  }
  if (currentGroup.length > 0) groups.push(currentGroup);

  return groups;
}

// ── Resolver Class ────────────────────────────────────────────────

export class SovereignEntityResolver {
  private modelRolodex: ModelRolodex;
  private cacheLimb: CacheForensicsLimb;
  private manifestMap: Map<number, ManifestEntry[]>;

  constructor(
    modelRolodex: ModelRolodex,
    cacheLimb: CacheForensicsLimb,
    manifestData: ManifestEntry[]
  ) {
    this.modelRolodex = modelRolodex;
    this.cacheLimb = cacheLimb;
    this.manifestMap = this.buildManifestMap(manifestData);
  }

  private buildManifestMap(data: ManifestEntry[]): Map<number, ManifestEntry[]> {
    const map = new Map<number, ManifestEntry[]>();
    const sorted = [...data].sort((a, b) => a._rawIndex - b._rawIndex);

    // Apply Sovereign Shorthand: collapse contiguous layers
    const groups = collapseLayers(sorted);

    for (const group of groups) {
      const head = group[0];
      if (!head.name) continue; // Skip orphan unnamed entries

      const id = head.id;
      map.set(id, group);
    }

    return map;
  }

  resolve(id: number, interactionContext?: string): ResolvedEntity | null {
    const group = this.manifestMap.get(id);
    if (!group) {
      // Fallback: cache forensics
      return this.resolveFromCache(id, interactionContext);
    }

    const head = group[0];
    const layers = group.slice(1);

    // Model versioning: resolve to latest if multiple share name
    const modelRef = this.resolveModelRef(head);

    // Variant mapping from layers
    const variants = this.extractVariants(group);

    // Interaction vocabulary filtering
    const activeContext = interactionContext
      ? INTERACTION_MAP[interactionContext] || 'ground'
      : 'ground';

    const filteredActions = this.filterActionsByContext(
      head.actions || [],
      activeContext
    );

    // Authentic classification
    const isAuthentic = AUTHENTIC_ANCHORS.has(head.name || '');

    // Bank-note detection
    const isNote = head.name?.toLowerCase().includes('note') || false;
    const category = this.classifyCategory(head, isNote, activeContext);

    return {
      id,
      canonicalName: head.name || `entity_${id}`,
      displayName: this.formatDisplayName(head.name, isNote),
      examine: head.examine || 'Nothing interesting happens.',
      category,
      modelRef,
      actions: filteredActions,
      variants,
      isAuthentic,
      layers,
      confidence: isAuthentic ? 1.0 : 0.85,
      groundingSource: 'manifest',
    };
  }

  private resolveModelRef(head: ManifestEntry): ResolvedEntity['modelRef'] {
    if (head.modelId) {
      return {
        id: head.modelId,
        name: head.modelName || `model_${head.modelId}`,
        version: head.version || 1,
      };
    }

    // Query ModelRolodex by name, resolve to latest version
    if (head.modelName) {
      const latest = this.modelRolodex.resolveLatest(head.modelName);
      if (latest) {
        return {
          id: latest.id,
          name: latest.name,
          version: latest.version,
        };
      }
    }

    // Final fallback
    return {
      id: head.modelId || 0,
      name: head.modelName || 'unknown',
      version: 1,
    };
  }

  private extractVariants(group: ManifestEntry[]): VariantSpec[] {
    return group
      .filter(e => e.actions && e.actions.length > 0)
      .map(e => ({
        context: this.inferContextFromActions(e.actions || []),
        modelId: e.modelId,
        actions: e.actions || [],
        isNote: e.name?.toLowerCase().includes('note') || false,
      }));
  }

  private inferContextFromActions(actions: string[]): VariantSpec['context'] {
    for (const action of actions) {
      const mapped = INTERACTION_MAP[action];
      if (mapped) return mapped;
    }
    return 'ground';
  }

  private filterActionsByContext(
    actions: string[],
    context: VariantSpec['context']
  ): string[] {
    const priorityActions = new Set([
      'Examine',
      'Use',
      'Cancel',
    ]);

    // Context-specific filtering
    const contextMap: Record<VariantSpec['context'], string[]> = {
      inventory: ['Use', 'Drop', 'Examine', 'Eat', 'Drink', 'Wield', 'Wear'],
      bank: ['Withdraw', 'Deposit', 'Examine', 'Cancel'],
      shop: ['Buy', 'Sell', 'Examine', 'Value'],
      trade: ['Offer', 'Remove', 'Accept', 'Decline', 'Examine'],
      wield: ['Remove', 'Examine'],
      ground: ['Take', 'Pick up', 'Examine', 'Use'],
    };

    const allowed = new Set([
      ...(contextMap[context] || []),
      ...priorityActions,
    ]);

    return actions.filter(a => allowed.has(a));
  }

  private classifyCategory(
    head: ManifestEntry,
    isNote: boolean,
    context: VariantSpec['context']
  ): ResolvedEntity['category'] {
    if (isNote) return 'bank_note';
    if (head.category === 'npc') return 'npc';
    if (head.category === 'item') {
      if (context === 'ground') return 'ground_item';
      return 'item';
    }
    if (head.category === 'object') return 'object';
    return 'object'; // Default
  }

  private formatDisplayName(name: string | undefined, isNote: boolean): string {
    if (!name) return 'Unknown';
    if (isNote && !name.toLowerCase().startsWith('note:')) {
      return `Note: ${name}`;
    }
    return name;
  }

  private resolveFromCache(
    id: number,
    interactionContext?: string
  ): ResolvedEntity | null {
    const cacheEntry = this.cacheLimb.lookup(id);
    if (!cacheEntry) return null;

    // Lower confidence for cache-only resolution
    return {
      id,
      canonicalName: cacheEntry.name || `cache_${id}`,
      displayName: cacheEntry.name || `Entity ${id}`,
      examine: cacheEntry.examine || 'Nothing interesting happens.',
      category: cacheEntry.category || 'object',
      modelRef: {
        id: cacheEntry.modelId || 0,
        name: `cache_model_${id}`,
        version: 1,
      },
      actions: cacheEntry.actions || ['Examine'],
      variants: [],
      isAuthentic: AUTHENTIC_ANCHORS.has(cacheEntry.name || ''),
      layers: [],
      confidence: 0.6,
      groundingSource: 'cache',
    };
  }

  // Batch resolution for predictive window (66x66 tile buffer)
  resolveBatch(ids: number[]): Map<number, ResolvedEntity> {
    const results = new Map<number, ResolvedEntity>();
    for (const id of ids) {
      const resolved = this.resolve(id);
      if (resolved) results.set(id, resolved);
    }
    return results;
  }
}
```

Connects resolver output to pedagogy corpus, enriches with learning notes.

```typescript
// D:\POG2\src\services\SovereignPedagogyBridge.ts

import { ResolvedEntity } from './SovereignEntityResolver';

// ── Types ──────────────────────────────────────────────────────────

export interface PedagogyEnrichment {
  learningNote: string;
  confidence: number; // Inherited from resolver, adjusted by pedagogy match
  suggestedAction: string;
  relatedEntities: number[]; // IDs of pedagogically linked entities
  pedagogicalTags: string[];
}

export interface EnrichedEntity extends ResolvedEntity {
  pedagogy: PedagogyEnrichment;
}

// ── Pedagogy Corpus Interface ──────────────────────────────────────

interface CorpusEntry {
  id: number;
  name: string;
  tags: string[];
  learningNote: string;
  suggestedActions: string[];
  relatedIds: number[];
}

// ── Bridge Class ───────────────────────────────────────────────────

export class SovereignPedagogyBridge {
  private corpus: Map<number, CorpusEntry>;
  private defaultEnrichment: PedagogyEnrichment;

  constructor(corpusData: CorpusEntry[]) {
    this.corpus = new Map(corpusData.map(e => [e.id, e]));
    this.defaultEnrichment = {
      learningNote: 'No pedagogical data available. Explore to discover properties.',
      confidence: 0.5,
      suggestedAction: 'Examine',
      relatedEntities: [],
      pedagogicalTags: ['unverified'],
    };
  }

  enrich(resolved: ResolvedEntity): EnrichedEntity {
    const corpusEntry = this.corpus.get(resolved.id);

    if (!corpusEntry) {
      // No direct match — try fuzzy name match or return default
      return {
        ...resolved,
        pedagogy: {
          ...this.defaultEnrichment,
          confidence: resolved.confidence * 0.7,
        },
      };
    }

    // Merge resolver confidence with pedagogy confidence
    const mergedConfidence = Math.min(
      1.0,
      (resolved.confidence + 0.9) / 2 // Corpus entries are high-confidence
    );

    // Suggested action: prioritize corpus, fallback to resolver actions
    const suggestedAction =
      corpusEntry.suggestedActions[0] ||
      resolved.actions[0] ||
      'Examine';

    return {
      ...resolved,
      pedagogy: {
        learningNote: corpusEntry.learningNote,
        confidence: mergedConfidence,
        suggestedAction,
        relatedEntities: corpusEntry.relatedIds,
        pedagogicalTags: [...corpusEntry.tags, resolved.groundingSource],
      },
    };
  }

  // Batch enrich for predictive window
  enrichBatch(resolvedMap: Map<number, ResolvedEntity>): Map<number, EnrichedEntity> {
    const enriched = new Map<number, EnrichedEntity>();
    for (const [id, resolved] of resolvedMap) {
      enriched.set(id, this.enrich(resolved));
    }
    return enriched;
  }
}
```

```typescript
import { ModelRolodex } from '../api/ai/ModelRolodex';
import { CacheForensicsLimb } from '../limbs/CacheForensicsLimb';

// ── Types ──────────────────────────────────────────────────────────

interface PedagogyRequest {
  type: 'REQUEST_PEDAGOGY';
  entityId: number;
  interactionContext?: string;
  playerPosition: { x: number; y: number; plane: number };
}

interface PedagogyResponse {
  type: 'ENTITY_DATA';
  entityId: number;
  data: EnrichedEntity | null;
  cached: boolean;
  timestamp: number;
}

// ── KV Cache Key Schema ────────────────────────────────────────────

function pedagogyCacheKey(entityId: number, context?: string): string {
  return `pedagogy:${entityId}${context ? `:${context}` : ''}`;
}

// ── Inside POG2WorldDO class ─────────────────────────────────────

export class POG2WorldDO {
  private resolver: SovereignEntityResolver;
  private bridge: SovereignPedagogyBridge;
  private modelRolodex: ModelRolodex;
  private cacheLimb: CacheForensicsLimb;
  private env: Env; // Cloudflare bindings

  constructor(state: DurableObjectState, env: Env) {
    this.env = env;

    // Initialize dependencies
    this.modelRolodex = new ModelRolodex();
    this.cacheLimb = new CacheForensicsLimb();

    // Load manifests from bundled assets
    const manifestData = this.loadManifests();
    this.resolver = new SovereignEntityResolver(
      this.modelRolodex,
      this.cacheLimb,
      manifestData
    );

    // Load pedagogy corpus
    const corpusData = this.loadPedagogyCorpus();
    this.bridge = new SovereignPedagogyBridge(corpusData);
  }

  private loadManifests(): ManifestEntry[] {
    // Manifests are bundled into the worker via import
    // Adjust path based on your bundler configuration
    const manifests = require('../assets/manifests/sovereign_manifests.json');
    return manifests.map((m: any, idx: number) => ({
      ...m,
      _rawIndex: idx,
    }));
  }

  private loadPedagogyCorpus(): CorpusEntry[] {
    // Corpus loaded from bundled JSON or KV
    // Prefer KV for runtime updates without redeploy
    // Fallback to bundled snapshot
    try {
      return require('../assets/pedagogy/corpus_snapshot.json');
    } catch {
      return [];
    }
  }

  async fetch(request: Request): Promise<Response> {
    const url = new URL(request.url);

    // Existing health check and routing
    if (url.pathname === '/health') {
      return new Response('OK', { status: 200 });
    }

    // WebSocket upgrade
    if (request.headers.get('Upgrade') === 'websocket') {
      return this.handleWebSocket(request);
    }

    return new Response('Not Found', { status: 404 });
  }

  private async handleWebSocket(request: Request): Promise<Response> {
    const [client, server] = Object.values(new WebSocketPair());
    server.accept();

    server.addEventListener('message', async (event) => {
      try {
        const msg = JSON.parse(event.data as string);

        if (msg.type === 'REQUEST_PEDAGOGY') {
          await this.handlePedagogyRequest(server, msg as PedagogyRequest);
        }
        // ... other message handlers
      } catch (err) {
        server.send(JSON.stringify({
          type: 'ERROR',
          error: err instanceof Error ? err.message : 'Unknown error',
        }));
      }
    });

    return new Response(null, { status: 101, webSocket: client });
  }

  private async handlePedagogyRequest(
    ws: WebSocket,
    req: PedagogyRequest
  ): Promise<void> {
    const cacheKey = pedagogyCacheKey(req.entityId, req.interactionContext);
    const now = Date.now();

    // 1. Check KV cache first
    const cached = await this.env.POG2_KV.get(cacheKey, 'json') as PedagogyResponse | null;
    if (cached && (now - cached.timestamp) < 300000) { // 5 min TTL
      ws.send(JSON.stringify({
        ...cached,
        cached: true,
      }));
      return;
    }

    // 2. Resolve entity
    const resolved = this.resolver.resolve(req.entityId, req.interactionContext);
    if (!resolved) {
      ws.send(JSON.stringify({
        type: 'ENTITY_DATA',
        entityId: req.entityId,
        data: null,
        cached: false,
        timestamp: now,
      }));
      return;
    }

    // 3. Enrich with pedagogy
    const enriched = this.bridge.enrich(resolved);

    // 4. Build response
    const response: PedagogyResponse = {
      type: 'ENTITY_DATA',
      entityId: req.entityId,
      data: enriched,
      cached: false,
      timestamp: now,
    };

    // 5. Cache in KV (async, non-blocking)
    this.env.POG2_KV.put(cacheKey, JSON.stringify(response), {
      expirationTtl: 300, // 5 minutes
    }).catch(err => {
      console.error(`KV cache write failed for ${cacheKey}:`, err);
    });

    // 6. Send to client
    ws.send(JSON.stringify(response));
  }

  // Predictive window: pre-resolve entities in 66x66 tile buffer
  async warmPredictiveWindow(
    centerX: number,
    centerY: number,
    plane: number
  ): Promise<void> {
    const halfSize = 33; // 66x66 centered
    const entityIds = await this.querySpatialIndex(
      centerX - halfSize,
      centerY - halfSize,
      centerX + halfSize,
      centerY + halfSize,
      plane
    );

    const resolved = this.resolver.resolveBatch(entityIds);
    const enriched = this.bridge.enrichBatch(resolved);

    // Pre-warm KV cache for all entities in window
    const batch = Array.from(enriched.entries()).map(([id, data]) => {
      const key = pedagogyCacheKey(id);
      return this.env.POG2_KV.put(key, JSON.stringify({
        type: 'ENTITY_DATA',
        entityId: id,
        data,
        cached: false,
        timestamp: Date.now(),
      }), { expirationTtl: 300 });
    });

    await Promise.all(batch);
  }
```

```tsx
    containerRef.current.appendChild(labelRenderer.domElement);

      <div class="hud-note">${data.pedagogy?.learningNote || 'No data'}</div>

      existing.element.style.cssText = labelDiv.style.cssText;

      label.position.set(data.position?.x || 0, data.position?.y || 0, data.position?.z || 0);

    mouseRef.current.y = -((event.clientY - rect.top) / rect.height) * 2 + 1;

      scene.children,
      true

        // Send pedagogy request

          type: 'REQUEST_PEDAGOGY',
          entityId,
          interactionContext: inferContextFromAction(hit.object.userData.lastAction),
          playerPosition: getPlayerPosition(), // From your store
        }));
      }
    } else {
      setHoveredEntity(null);
    }
  }, [camera, scene, renderer, websocket, hoveredEntity, setHoveredEntity]);

  useEffect(() => {
    renderer.domElement.addEventListener('mousemove', handleMouseMove);
    return () => renderer.domElement.removeEventListener('mousemove', handleMouseMove);
  }, [handleMouseMove, renderer]);

  // Cleanup on unmount
  useEffect(() => {
    return () => {
      labelsRef.current.forEach(label => scene.remove(label));
      labelsRef.current.clear();
    };
  }, [scene]);

  return <div ref={containerRef} style={{ position: 'absolute', inset: 0 }} />;
};
```

## 5. ModelRolodex.ts — Latest Version Resolution

```typescript
// D:\POG2\src\api\ai\ModelRolodex.ts

interface ModelEntry {
  id: number;
  name: string;
  version: number;
  timestamp: number;
  metadata?: Record<string, any>;
}

export class ModelRolodex {
  private modelsByName: Map<string, ModelEntry[]>;
  private modelsById: Map<number, ModelEntry>;

  constructor() {
    this.modelsByName = new Map();
    this.modelsById = new Map();
  }

  register(entry: ModelEntry): void {
    this.modelsById.set(entry.id, entry);

    const existing = this.modelsByName.get(entry.name) || [];
    existing.push(entry);
    this.modelsByName.set(entry.name, existing);
  }

  // Sovereign Truth: resolve to latest version for name collisions
  resolveLatest(name: string): ModelEntry | null {
    const entries = this.modelsByName.get(name);
    if (!entries || entries.length === 0) return null;

    // Sort by version descending, then timestamp descending
    const sorted = [...entries].sort((a, b) => {
      if (b.version !== a.version) return b.version - a.version;
      return b.timestamp - a.timestamp;
    });

    return sorted[0];
  }

  resolveById(id: number): ModelEntry | null {
    return this.modelsById.get(id) || null;
  }

  // Batch load from cache entities (229k entries)
  loadFromCacheEntities(entities: Array<{ id: number; name: string; modelId: number; version?: number }>): void {
    for (const e of entities) {
      this.register({
        id: e.modelId,
        name: e.name,
        version: e.version || 1,
        timestamp: Date.now(),
      });
    }
  }
}
```
