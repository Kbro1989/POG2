## Custom Shader Approach (Recommended)

```typescript
const material = new THREE.ShaderMaterial({
    uniforms: {
        uTime: { value: 0 }
    },
    vertexShader: `
        uniform float uTime;
        attribute float aPhase;
        attribute vec3 aBaseScale;
        varying vec3 vColor;
        
        void main() {
            float pulse = sin(uTime * 3.0 + aPhase) * 0.2 + 1.0;
            vec3 scaled = position * aBaseScale * pulse;
            gl_Position = projectionMatrix * modelViewMatrix * instanceMatrix * vec4(scaled, 1.0);
            vColor = instanceColor;
        }
    `,
    fragmentShader: `
        varying vec3 vColor;
        void main() {
            gl_FragColor = vec4(vColor, 0.6);
        }
    `,
    transparent: true,
    depthTest: false,
    blending: THREE.AdditiveBlending
});
```

This gives **individual pulsation** without CPU matrix updates every frame.

## Corrected `getHeatAtPosition`

```typescript
private getHeatAtPosition(snapshot: GhostSplatSnapshot, entity: any): number {
    if (entity.sovereignHeat !== undefined) return entity.sovereignHeat;
    
    // Convert world position to mapsquare-local tile coords
    const [msX, msZ] = snapshot.mapsquare.split(',').map(Number);
    const msWorldX = msX * 64 * this.TILE_SIZE;
    const msWorldZ = msZ * 64 * this.TILE_SIZE;
    
    const localX = Math.floor((entity.position.x - msWorldX) / this.TILE_SIZE);
    const localZ = Math.floor((entity.position.z - msWorldZ) / this.TILE_SIZE);
    
    // Clamp to 0-65 (66×66 window)
    const lx = Math.max(0, Math.min(65, localX));
    const lz = Math.max(0, Math.min(65, localZ));
    
    const idx = lz * 66 + lx;
    return snapshot.heatmap[idx] || 0;
}
```
