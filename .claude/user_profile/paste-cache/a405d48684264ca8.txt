// WikiEquipmentEnricher.ts — update enrichItem()
public async enrichItem(name: string): Promise<WikiPageData> {
    const cached = this.readCache('items', name);
    if (cached?.title) return cached;

    // 1. Get wiki data (static)
    const bonuses = await this.parseEquipmentBonuses(name);
    const itemMeta = await this.parseItemMeta(name);
    
    // 2. Get canonical ID from cache
    const resolution = await CacheForensicsLimb.resolveWikiTitle(name);
    
    // 3. Get live GE price (NEW)
    let gePrice: GePrice | null = null;
    if (resolution.primaryId) {
        gePrice = await GeApi.getLatestPrice(resolution.primaryId);
    }
    
    // 4. Merge — live price overrides wiki static price
    if (gePrice && itemMeta) {
        itemMeta.geValue = gePrice.price;  // Real-time override
        itemMeta._geSource = gePrice.source;
        itemMeta._geTimestamp = gePrice.timestamp;
    }

    // ... rest of enrichment
}
