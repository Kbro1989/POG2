## Recommended Unification

**Extend `WikiEquipmentEnricher` with a generic `enrichPage()` method** that handles any page type (place, hitsplat, mechanic, lore):

```typescript
// Add to WikiEquipmentEnricher.ts
export interface WikiGenericMeta {
    title: string;
    pageType: 'npc' | 'item' | 'equipment' | 'place' | 'mechanic' | 'lore';
    extract: string;
    infoboxVars: Record<string, string>;
    categoryMembers?: string[];  // For image categories like Category:Hitsplat_images
    fetchedAt: number;
}

public async enrichGeneric(pageTitle: string): Promise<<WikiPageData> {
    const cached = this.readCache('generic', pageTitle);
    if (cached?.title) return cached;

    const wikitext = await this.getWikitextForName(pageTitle);
    if (!wikitext) return null;

    // Detect page type from infobox or categories
    const pageType = this.detectPageType(wikitext);
    
    // Extract category members if it's a category page
    let categoryMembers: string[] | undefined;
    if (pageTitle.startsWith('Category:')) {
        categoryMembers = await this.fetchCategoryMembers(pageTitle);
    }

    const result: WikiPageData = {
        title: pageTitle,
        fetchedAt: Date.now(),
        bonuses: null,
        itemMeta: null,
        monsterMeta: null,
        dropTables: [],
        rawVariables: this.extractAllVariables(wikitext),
        // Extend WikiPageData with generic fields
        pageType,
        extract: await this.fetchExtract(pageTitle),
        categoryMembers
    } as any;

    this.writeCache('generic', pageTitle, result);
    return result;
}
```
