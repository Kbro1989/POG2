## Recommended: Function Calling + Fallback

```typescript
export class SovereignPedagogyBridge {
    private rolodex: ModelRolodex;
    private kv: KVNamespace;

    constructor(env: { ROLodex: ModelRolodex; KV: KVNamespace }) {
        this.rolodex = env.ROLodex;
        this.kv = env.KV;
    }

    async enrich(entity: RawEntityData): Promise<PedagogyResult> {
        const cacheKey = `pedagogy:${entity.name}:${entity.type}`;
        
        // 1. Check KV cache
        const cached = await this.kv.get(cacheKey);
        if (cached) return JSON.parse(cached);

        // 2. Query model with function calling
        const result = await this.queryWithStructure(entity);

        // 3. Validate + clamp
        const validated = this.validate(result);

        // 4. Cache and return
        await this.kv.put(cacheKey, JSON.stringify(validated), { expirationTtl: 3600 });
        return validated;
    }

    private async queryWithStructure(entity: RawEntityData): Promise<PedagogyResult> {
        const prompt = this.buildPrompt(entity);
        
        const response = await this.rolodex.query({
            model: 'pog2-pedagogy',
            prompt,
            tools: [this.pedagogyToolSchema],
            toolChoice: {type: 'function', function: {name: 'emit_pedagogy_result'}}
        });

        // Extract function call arguments
        const args = response.toolCalls?.[0]?.function?.arguments;
        if (args) return JSON.parse(args);

        // Fallback: parse from text
        return this.fallbackParse(response.text);
    }

    private validate(result: PedagogyResult): PedagogyResult {
        return {
            learningNote: result.learningNote || 'No pedagogical insight available.',
            confidence: Math.max(0, Math.min(1, result.confidence || 0.5)),
            suggestedAction: result.suggestedAction || 'Observe and record.',
            emotionalWeight: this.normalizeWeights(result.emotionalWeight),
            hexagramPin: {
                id: Math.max(1, Math.min(64, result.hexagramPin?.id || 1)),
                name: result.hexagramPin?.name || 'The Creative',
                line: Math.max(1, Math.min(6, result.hexagramPin?.line || 1))
            }
        };
    }

    private normalizeWeights(weights: Record<string, number>): Record<string, number> {
        const normalized: Record<string, number> = {};
        for (const [k, v] of Object.entries(weights || {})) {
            normalized[k] = Math.max(0, Math.min(1, v));
        }
        return normalized;
    }
}
```
