// In generate_pedagogy_corpus.ts
import * as fs from 'fs';

const ITEMS_DUMP = 'D:/D-sovereign-/cache_pedagogy/json_dumps/items.json';
const NPCS_DUMP = 'D:/D-sovereign-/cache_pedagogy/json_dumps/npcs.json';

async function generateFullCorpus() {
    const items = JSON.parse(fs.readFileSync(ITEMS_DUMP, 'utf8'));
    const npcs = JSON.parse(fs.readFileSync(NPCS_DUMP, 'utf8'));
    
    const corpus = [];
    
    // Process items (batch for memory)
    for (let i = 0; i < items.length; i += 100) {
        const batch = items.slice(i, i + 100);
        for (const item of batch) {
            const enriched = await enrichItem(item.name);
            corpus.push(enriched);
        }
        // Write incremental to avoid memory bloat
        appendToCorpus(corpus.splice(0, corpus.length));
    }
    
    // Process NPCs
    for (let i = 0; i < npcs.length; i += 100) {
        const batch = npcs.slice(i, i + 100);
        for (const npc of batch) {
            const enriched = await enrichMonster(npc.name);
            corpus.push(enriched);
        }
        appendToCorpus(corpus.splice(0, corpus.length));
    }
}

function appendToCorpus(entries: any[]) {
    const lines = entries.map(e => JSON.stringify(e)).join('\n');
    fs.appendFileSync('D:/sovereign/memory/pedagogy/corpus/verified_entities.jsonl', lines + '\n');
}
