// In generate():
try {
    logger.info('Processing Mechanics Pages...');
    const results = [];
    for (const page of MECHANICS_PAGES) {
        const data = await enricher.enrichGeneric(page);
        if (data) {
            results.push({
                instruction: `Explain the ${page} mechanic in RS3`,
                input: JSON.stringify({ mechanic: page, wiki: data }),
                output: formatVerifiedOutput(page, 0, data),
                tags: ['[WIKI_ONLY]', '[MECHANIC]'],
                verified: false
            });
        }
    }
    appendToCorpus(results);
} catch (err) {
    logger.error({ err }, 'Mechanics corpus generation failed');
}
