```typescript
// Test script: POG2/scripts/test_query_classifier.ts
import { QueryClassifier } from '../src/utils/QueryClassifier.js';

const tests = [
  { input: 'enum 1234', expected: 'structural_enum', cacheOnly: true },
  { input: 'varbit 28556', expected: 'structural_varbit', cacheOnly: true },
  { input: 'dbrow 5678', expected: 'structural_dbrow', cacheOnly: true },
  { input: '50,50', expected: 'spatial_area', cacheOnly: true },
  { input: 'Lumbridge Castle', expected: 'semantic_name', cacheOnly: false },
  { input: 'Abyssal demon 1615', expected: 'hybrid', cacheOnly: false },
  { input: '12345', expected: 'structural_id', cacheOnly: true },  // Pure number

for (const t of tests) {
  const result = QueryClassifier.classify(t.input);
  console.assert(result.type === t.expected, `FAIL: ${t.input} → ${result.type} (expected ${t.expected})`);
  console.assert(result.cacheOnly === t.cacheOnly, `FAIL: ${t.input} cacheOnly mismatch`);
}
console.log('QueryClassifier: All tests passed');
```

**Expected:** 7/7 pass. The pure number case (`12345`) is the most ambiguous — verify it routes to `structural_id` and triggers `CacheForensicsLimb` first, not wiki search.

### 6. Pedagogy Tag Accuracy

```typescript
// Verify training examples have correct tags
const corpus = JSON.parse(fs.readFileSync('pog2_pedagogy_corpus.jsonl', 'utf8')
  .split('\n')
  .filter(Boolean)
  .map(JSON.parse));

const tags = corpus.map((e: any) => {
  const output = e.output as string;
  if (output.includes('[CACHE_VERIFIED]')) return 'cache';
  if (output.includes('[VERIFIED]')) return 'verified';
  if (output.includes('[WIKI_ONLY]')) return 'wiki';
  if (output.includes('[REMOVED]')) return 'removed';
  return 'untagged';
});

const counts = tags.reduce((acc: any, t: string) => { acc[t] = (acc[t] || 0) + 1; return acc; }, {});
console.log('Tag distribution:', counts);

// Verify no structural queries are tagged [VERIFIED] (should be [CACHE_VERIFIED])
const structuralVerified = corpus.filter((e: any) => 
  e.input.includes('structural_') && e.output.includes('[VERIFIED]')
);
console.assert(structuralVerified.length === 0, 
  `FAIL: ${structuralVerified.length} structural entries incorrectly tagged [VERIFIED]`);
```

**Expected:** 
- `cache` tags dominate for structural queries
- `verified` tags for hybrid/API+wiki matches
- `wiki` tags for lore-only entries
- `removed` tags for historical content
- Zero structural entries with `[VERIFIED]` — that's a routing bug
