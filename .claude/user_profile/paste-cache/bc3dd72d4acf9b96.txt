import { ResolvedEntity } from './SovereignEntityResolver';

// ── Note Templates ─────────────────────────────────────────────────

const NOTE_TEMPLATES: Record<string, (e: ResolvedEntity, wiki?: WikiEnrichment) => string> = {
  npc: (e, w) => {
    const parts: string[] = [];
    if (w?.levelRequirement) parts.push(`Combat level ${w.levelRequirement}`);
    if (w?.combatStats?.weakness) parts.push(`Weak to ${w.combatStats.weakness}`);
    if (w?.combatStats?.aggression) parts.push('Aggressive');
    if (e.isAuthentic) parts.push('Authentic Anchor — cross-map consistent');

  },
  item: (e, w) => {

// ── Action Ranker ─────────────────────────────────────────────────

function rankSuggestedAction(entity: ResolvedEntity, wiki?: WikiEnrichment): { action: string; rationale: string } {
  // Priority 1: Wiki-informed combat
  if (wiki?.combatStats && entity.actions.includes('Attack')) {
    return {
      action: 'Attack',
      rationale: `Combat level ${wiki.levelRequirement}. Aggressive: ${wiki.combatStats.aggression}.`,
    };
  }

  // Priority 2: Quest NPC
  if (wiki?.questRequirement && entity.actions.includes('Talk to')) {
    return { action: 'Talk to', rationale: `Quest: ${wiki.questRequirement}` };
  }

  // Priority 3: Authentic anchors
  if (entity.isAuthentic) {
    if (entity.actions.includes('Bank')) return { action: 'Bank', rationale: 'Authentic banking anchor' };
    if (entity.actions.includes('Trade')) return { action: 'Trade', rationale: 'Authentic trade anchor' };
    if (entity.actions.includes('Talk to')) return { action: 'Talk to', rationale: 'Authentic dialogue anchor' };
  }

  // Priority 4: Frequency-weighted from actionFrequencies
  let bestAction = 'Examine';
  let bestFreq = 0;
  for (const [action, freq] of entity.actionFrequencies) {
    if (freq > bestFreq) {
      bestFreq = freq;
      bestAction = action;
    }
  }

  // Context overrides
  if (entity.category === 'bank_note' && entity.actions.includes('Withdraw')) {
    return { action: 'Withdraw', rationale: 'Convert note to physical item' };
  }
  if (entity.category === 'ground_item' && entity.actions.includes('Take')) {
    return { action: 'Take', rationale: 'Primary ground acquisition' };
  }

  return { action: bestAction, rationale: `Highest corpus frequency: ${bestFreq}` };
}

export class SovereignPedagogyBridge {
  private wikiEnricher: WikiEquipmentEnricher;
  private bestiaryApi: BestiaryApi;
  private geApi: GeApi;
  private wikiCache: Map<number, WikiEnrichment>;
  private enrichmentCache: Map<number, EnrichedEntity>;

  constructor(wikiEnricher: WikiEquipmentEnricher, bestiaryApi: BestiaryApi, geApi: GeApi) {
    this.wikiEnricher = wikiEnricher;
    this.bestiaryApi = bestiaryApi;
    this.geApi = geApi;
    this.wikiCache = new Map();
    this.enrichmentCache = new Map();
  }

  async enrich(resolved: ResolvedEntity): Promise<EnrichedEntity> {
    const cached = this.enrichmentCache.get(resolved.id);
    if (cached) return cached;

    const wiki = await this.fetchWikiEnrichment(resolved);
    const { action, rationale } = rankSuggestedAction(resolved, wiki);
    const template = NOTE_TEMPLATES[resolved.category] || NOTE_TEMPLATES.object;
    
    const enriched: EnrichedEntity = {
      ...resolved,
      pedagogy: {
        description: wiki?.wikiTitle || resolved.canonicalName,
        examine: wiki?.examineText || resolved.examine,
        learningNote: template(resolved, wiki),
        confidence: this.computeConfidence(resolved, wiki),
        suggestedAction: action,
        suggestedActionRationale: rationale,
        relatedEntities: await this.findRelated(resolved, wiki),
        pedagogicalTags: this.buildTags(resolved, wiki),
        emotionalWeight: this.computeEmotionalWeight(resolved, wiki),
        hexagramPin: this.computeHexagramPin(resolved),
        wikiEnrichment: wiki,
      },
    };

    this.enrichmentCache.set(resolved.id, enriched);
    return enriched;
  }

  private async fetchWikiEnrichment(resolved: ResolvedEntity): Promise<WikiEnrichment | undefined> {
    if (this.wikiCache.has(resolved.id)) return this.wikiCache.get(resolved.id);

    try {
      let wiki: WikiEnrichment | undefined;

      if (resolved.category === 'npc') {
        const data = await this.bestiaryApi.fetch(resolved.id);
        if (data) wiki = this.transformBestiary(data);
      } else if (['item', 'ground_item', 'bank_note'].includes(resolved.category)) {
        const [geData, wikiItem] = await Promise.all([
          this.geApi.fetch(resolved.id).catch(() => null),
          this.wikiEnricher.enrichByName(resolved.canonicalName).catch(() => null),
        ]);
        wiki = this.mergeGeWiki(geData, wikiItem);
      } else {
        wiki = await this.wikiEnricher.enrichByName(resolved.canonicalName).catch(() => undefined);
      }

      if (wiki) this.wikiCache.set(resolved.id, wiki);
      return wiki;
    } catch (err) {
      console.warn(`Wiki fetch failed for ${resolved.id}:`, err);
      return undefined;
    }
  }

  private transformBestiary(data: any): WikiEnrichment {
    return {
      wikiUrl: data.wikiUrl || '',
      wikiTitle: data.name || '',
      examineText: data.examine || '',
      releaseDate: data.releaseDate,
      updateHistory: data.updateHistory || [],
      membersOnly: data.membersOnly || false,
      questRequirement: data.questRequired,
      levelRequirement: data.level,
      skillRequirements: data.skillRequirements || {},
      combatStats: data.combatStats ? {
        attackStyle: data.combatStats.attackStyle,
        weakness: data.combatStats.weakness,
        maxHit: data.combatStats.maxHit,
        aggression: data.combatStats.aggressive,
        poisonous: data.combatStats.poisonous,
      } : undefined,
      dropTable: data.drops?.map((d: any) => ({
        itemName: d.name,
        quantity: d.quantity,
        rarity: d.rarity,
        geValue: d.geValue,
      })),
      aliases: data.aliases || [],
      images: data.images || [],
    };
  }

  private mergeGeWiki(geData: any, wikiItem: any): WikiEnrichment {
    return {
      wikiUrl: wikiItem?.wikiUrl || '',
      wikiTitle: wikiItem?.name || geData?.name || '',
      examineText: wikiItem?.examine || geData?.examine || '',
      releaseDate: wikiItem?.releaseDate,
      updateHistory: wikiItem?.updateHistory || [],
      membersOnly: wikiItem?.membersOnly || false,
      questRequirement: wikiItem?.questRequired,
      levelRequirement: wikiItem?.levelRequirement,
      skillRequirements: wikiItem?.skillRequirements || {},
      gePrice: geData?.price,
      geVolume: geData?.volume,
      aliases: wikiItem?.aliases || [],
      images: wikiItem?.images || [],
    };
  }

  private computeConfidence(resolved: ResolvedEntity, wiki?: WikiEnrichment): number {
    let c = resolved.confidence;
    if (wiki) c += 0.05;
    if (wiki?.combatStats) c += 0.03;
    if (wiki?.dropTable) c += 0.02;
    return Math.min(c, 1.0);
  }

  private async findRelated(resolved: ResolvedEntity, wiki?: WikiEnrichment): Promise<number[]> {
    const related = new Set<number>();
    if (wiki?.dropTable) {
      for (const drop of wiki.dropTable) {
        const id = await this.wikiEnricher.resolveNameToId(drop.itemName).catch(() => null);
        if (id) related.add(id);
      }
    }
    return Array.from(related).slice(0, 10);
  }

  private buildTags(resolved: ResolvedEntity, wiki?: WikiEnrichment): string[] {
    const tags = new Set([resolved.groundingSource, resolved.category]);
    if (resolved.isAuthentic) tags.add('authentic');
    if (wiki) tags.add('wiki_enriched');
    if (wiki?.combatStats?.aggression) tags.add('aggressive');
    if (wiki?.gePrice) tags.add('tradeable');
    if (resolved.category === 'bank_note') tags.add('note');
    return Array.from(tags);
  }

  private computeEmotionalWeight(resolved: ResolvedEntity, wiki?: WikiEnrichment): Record<string, number> {
    const weight: Record<string, number> = {};
    if (resolved.isAuthentic) weight.familiarity = 0.9;
    if (wiki?.combatStats?.aggression) weight.threat = 0.7;
    if (wiki?.gePrice && wiki.gePrice > 1000000) weight.value = 0.8;
    if (resolved.category === 'bank_note') weight.convenience = 0.6;
    return weight;
  }

  private computeHexagramPin(resolved: ResolvedEntity): { id: number; name: string; line: number } {
    // Map entity properties to hexagram state
    // This is a placeholder — your HexagramManager would provide real mapping
    const hash = resolved.id % 64;
    return {
      id: hash + 1,
      name: `Hexagram ${hash + 1}`,
      line: resolved.confidence > 0.9 ? 6 : resolved.confidence > 0.7 ? 4 : 2,
    };
  }

  async enrichBatch(resolvedMap: Map<number, ResolvedEntity>, concurrency = 5): Promise<Map<number, EnrichedEntity>> {
    const enriched = new Map<number, EnrichedEntity>();
    const entries = Array.from(resolvedMap.entries());
    
    for (let i = 0; i < entries.length; i += concurrency) {
      const chunk = entries.slice(i, i + concurrency);
      await Promise.all(chunk.map(async ([id, resolved]) => {
        enriched.set(id, await this.enrich(resolved));
      }));
    }
    
    return enriched;
  }

  clearCaches(): void {
    this.wikiCache.clear();
    this.enrichmentCache.clear();
  }
}

// In your POG2WorldDO constructor or initialization:

import { SovereignEntityResolver } from '../services/SovereignEntityResolver';
import { SovereignPedagogyBridge } from '../services/SovereignPedagogyBridge';
import { WikiEquipmentEnricher } from '../api/wiki/WikiEquipmentEnricher';
import { BestiaryApi } from '../api/bestiary/BestiaryApi';
import { GeApi } from '../api/ge/GeApi';

export class POG2WorldDO {
  private resolver: SovereignEntityResolver;
  private bridge: SovereignPedagogyBridge;

  constructor(state: DurableObjectState, env: Env) {
    // Initialize resolver (singleton pattern)
    this.resolver = SovereignEntityResolver.getInstance();
    this.resolver.initialize(env);

    // Initialize bridge with API dependencies
    const wikiEnricher = new WikiEquipmentEnricher(env);
    const bestiaryApi = new BestiaryApi(env);
    const geApi = new GeApi(env);
    
    this.bridge = new SovereignPedagogyBridge(wikiEnricher, bestiaryApi, geApi);
  }

  // In your WebSocket handler:
  private async handlePedagogyRequest(ws: WebSocket, req: PedagogyRequest): Promise<void> {
    const cacheKey = `pedagogy:${req.entityId}:${req.interactionContext || 'default'}`;
    
    // Check KV
    const cached = await this.env.POG2_KV.get(cacheKey, 'json');
    if (cached) {
      ws.send(JSON.stringify({ ...cached, cached: true }));
      return;
    }

    // Resolve
    const result = await this.resolver.resolve(req.entityId, req.interactionContext);
    if (!result.ok || !result.value) {
      ws.send(JSON.stringify({ type: 'ENTITY_DATA', entityId: req.entityId, data: null }));
      return;
    }

    // Enrich (async wiki lookup)
    const enriched = await this.bridge.enrich(result.value);

    const response = {
      type: 'ENTITY_DATA',
      entityId: req.entityId,
      data: enriched,
      cached: false,
      timestamp: Date.now(),
    };

    // Cache and send
    await this.env.POG2_KV.put(cacheKey, JSON.stringify(response), { expirationTtl: 300 });
    ws.send(JSON.stringify(response));
  }
}
