## Corpus Generation Script — Key Implementation

The script should produce `.jsonl` format with proper tagging:

```typescript
// generate_pedagogy_corpus.ts — core logic
async function generateMonsterCorpus(names: string[]) {
    const examples = [];
    
    for (const name of names) {
        // 1. Bestiary API (canonical)
        const bestiary = await BestiaryApi.enrichByName(name);
        
        // 2. Wiki enrichment (semantic)
        const wiki = await WikiEquipmentEnricher.enrichMonster(name);
        
        // 3. Cache cross-reference
        const cache = await CacheForensicsLimb.resolveWikiTitle(name);
        
        // 4. Tag based on data quality
        const tags = [];
        if (bestiary) tags.push('[VERIFIED]');
        else tags.push('[WIKI_ONLY]');
        
        if (bestiary?._membersOverride) tags.push('[MEMBERS_OVERRIDE]');
        
        // 5. Format output
        examples.push({
            instruction: `Generate verified entity definition for ${name}`,
            input: JSON.stringify({
                bestiary: bestiary || null,
                wiki: wiki.monsterMeta || null,
                cache: cache.confidence !== 'none' ? cache : null
            }),
            output: formatMonsterOutput(bestiary, wiki, cache, tags),
            tags,
            verified: !!bestiary
        });
    }
    
    return examples;
}
```
