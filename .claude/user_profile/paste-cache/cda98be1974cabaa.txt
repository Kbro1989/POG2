import requests, json, asyncio, aiohttp
from pathlib import Path

BASE = "http://services.runescape.com/m=itemdb_rs/bestiary"
OUTPUT = Path("pog2_pedagogy_bestiary.jsonl")

async def fetch_beast(session, beast_id: int):
    """Primary source: Jagex Bestiary API. Cross-reference with cache entity ID."""
    url = f"{BASE}/beastData.json?beastid={beast_id}"
    async with session.get(url, timeout=30) as resp:
        if resp.status != 200:
            return None
        data = await resp.json()
        
        # Canonical fields per API spec
        canonical = {
            "id": data.get("id"),
            "name": data.get("name"),
            "examine": data.get("description"),  # Jagex calls it 'description'
            "combat_level": data.get("level"),
            "lifepoints": data.get("lifepoints"),
            "attack": data.get("attack"),
            "defence": data.get("defence"),
            "magic": data.get("magic"),
            "ranged": data.get("ranged"),
            "weakness": data.get("weakness"),
            "slayer_level": data.get("slayerlevel"),
            "slayer_cat": data.get("slayercat"),
            "size": data.get("size"),
            "aggressive": data.get("aggressive"),
            "poisonous": data.get("poisonous"),
            "areas": data.get("areas", []),
            "animations": data.get("animations", {}),
            "members": data.get("members"),
            "xp": data.get("xp"),
        }
        
        # Cross-reference with your cache entity (if exists in 229k pedagogy)
        cache_entity = lookup_cache_entity(beast_id)  # Your existing function
        
        return {
            "instruction": f"Generate a verified POG2 entity definition for beast ID {beast_id}",
            "input": json.dumps(canonical, indent=2),
            "output": format_pog2_output(canonical, cache_entity),
            "source": "jagex_bestiary_api",
            "verified": True,
            "cache_cross_reference": cache_entity is not None
        }

def format_pog2_output(bestiary_data, cache_entity):
    """Format as POG2 pedagogy with [VERIFIED] tags and limb routing hints."""
    lines = [
        f"[VERIFIED] Entity {bestiary_data['id']}: {bestiary_data['name']}",
        f"[SOURCE] Jagex Bestiary API + CacheForensicsLimb",
        "",
        f"## Combat Profile (GhostSplat Engine)",
        f"- Combat Level: {bestiary_data['combat_level']}",
        f"- Lifepoints: {bestiary_data['lifepoints']}",
        f"- Attack: {bestiary_data['attack']} | Defence: {bestiary_data['defence']}",
        f"- Magic: {bestiary_data['magic']} | Ranged: {bestiary_data['ranged']}",
        f"- Weakness: {bestiary_data['weakness']} (ID: {weakness_id_map.get(bestiary_data['weakness'], 'N/A')})",
        f"- Slayer Requirement: {bestiary_data['slayer_level'] or 'None'}",
        f"- Slayer Category: {bestiary_data['slayer_cat'] or 'None'}",
        f"- Size: {bestiary_data['size']}x{bestiary_data['size']}",
        f"- Aggressive: {bestiary_data['aggressive']} | Poisonous: {bestiary_data['poisonous']}",
        "",
        f"## Spatial Profile (SpatialPipeline)",
        f"- Spawn Areas: {', '.join(bestiary_data['areas'])}",
        f"- Animation Keys: {json.dumps(bestiary_data['animations'])}",
        "",
        f"## Examine Text",
        f'"{bestiary_data["examine"]}"',
    ]
    
    if cache_entity:
        lines.extend([
            "",
            f"## Cache Cross-Reference (CacheForensicsLimb)",
            f"- Cache Entity ID: {cache_entity.get('id')}",
            f"- Cache Revision: {cache_entity.get('revision', 'unknown')}",
            f"- Varbit Masks: {cache_entity.get('varbits', [])}",
            f"- Collision Layer: {cache_entity.get('collision', 'unverified')}",
        ])
    
    lines.extend([
        "",
        "## Hexagram Routing",
        "- Primary Limb: CombatRotationEngine",
        "- Secondary: LootResolutionEngine",
        "- Fallback: AggressionSimulator",
        "- Yao Line 3 (Action): Engage if combat_level <= player_level + 10",
    ])
    
    return "\n".join(lines)

# Batch scrape all known beast IDs (0-20000 range, filter valid responses)
async def build_corpus():
    async with aiohttp.ClientSession() as session:
        tasks = [fetch_beast(session, i) for i in range(20000)]
        results = await asyncio.gather(*tasks)
        
    valid = [r for r in results if r]
    with open(OUTPUT, "w") as f:
        for entry in valid:
            f.write(json.dumps(entry) + "\n")
    
    print(f"Corpus built: {len(valid)} verified entities")

## 2. Local Ollama: Unsloth LoRA → Adapter + Merged GGUF

Your `deepseek-r1:8b` and `qwen3-coder` local models are the base. Train with Unsloth for 80% VRAM reduction, then export both LoRA adapter and merged GGUF.

```python
from unsloth import FastLanguageModel
from transformers import TrainingArguments
from trl import SFTTrainer
import torch, json
from datasets import load_dataset

# Load base — qwen3-coder optimized for structured data/code
model, tokenizer = FastLanguageModel.from_pretrained(
    model_name = "unsloth/Qwen3-8B",
    max_seq_length = 32768,  # Bestiary entries + cache cross-references need room
    dtype = torch.bfloat16,
    load_in_4bit = True,
)

model = FastLanguageModel.get_peft_model(
    model,
    r = 64,                # High rank for complex game logic
    target_modules = ["q_proj", "k_proj", "v_proj", "o_proj",
                      "gate_proj", "up_proj", "down_proj"],
    lora_alpha = 128,
    lora_dropout = 0,
    bias = "none",
    use_gradient_checkpointing = "unsloth",
    random_state = 3407,
    use_rslora = True,     # Rank-stabilized LoRA for better convergence
)

# POG2-specific prompt template embedding 27-limb ontology
pog2_template = """You are the POG2 Pedagogy Engine, trained on 229,000 verified RuneScape cache entities and Jagex Bestiary API primary sources.

Your 27-limb neural architecture includes:
- Creative Core: image, animation, material, meshops, video, creative, audio
- Technical Core: code, data, file, entity, system, versioncontrol, network, security
- Spatial Core: spatialpipeline, landscapegeneration, physics, world, livegame
- Metaphysical Core: ghost, divine, quantum, reality, relic, idauditor, orchestrator

Rules:
1. [VERIFIED] = confirmed by Jagex API + cache forensics
2. [UNVERIFIED] = wiki or community source, flag uncertainty
3. [WIKI_CONFLICT] = discrepancy between API and wiki, note both
4. Always reference entity IDs (NPC 239, Item 4151, etc.)
5. Use POG2 limb terminology for routing hints
6. Never hallucinate stats — if unknown, state [DATA_MISSING]

### Instruction:
{}

### Input:
{}

### Response:
{}"""

EOS_TOKEN = tokenizer.eos_token

def format_samples(examples):
    instructions = examples["instruction"]
    inputs = examples["input"]
    outputs = examples["output"]
    texts = []
    for i, inp, out in zip(instructions, inputs, outputs):
        text = pog2_template.format(i, inp, out) + EOS_TOKEN
        texts.append(text)
    return {"text": texts}

dataset = load_dataset("json", data_files="pog2_pedagogy_bestiary.jsonl", split="train")
dataset = dataset.map(format_samples, batched=True)

trainer = SFTTrainer(
    model = model,
    tokenizer = tokenizer,
    train_dataset = dataset,
    dataset_text_field = "text",
    max_seq_length = 32768,
    dataset_num_proc = 4,
    packing = True,  # Efficient packing for variable-length entries
    args = TrainingArguments(
        per_device_train_batch_size = 1,
        gradient_accumulation_steps = 8,
        warmup_steps = 10,
        max_steps = 1000,
        learning_rate = 1e-4,
        fp16 = not torch.cuda.is_bf16_supported(),
        bf16 = torch.cuda.is_bf16_supported(),
        logging_steps = 1,
        optim = "adamw_8bit",
        weight_decay = 0.01,
        lr_scheduler_type = "cosine",
        seed = 3407,
        output_dir = "pog2_pedagogy_outputs",
        report_to = "none",
    ),
)

trainer.train()

# Save LoRA adapter for Ollama ADAPTER directive
model.save_pretrained("pog2_pedagogy_lora")
tokenizer.save_pretrained("pog2_pedagogy_lora")

# Export merged GGUF for zero-overhead inference
# Q8_0 for quality (combat stats must be precise), Q4_K_M for GhostLimb fallback
model.save_pretrained_gguf("pog2_pedagogy_q8", tokenizer, quantization_method="q8_0")
model.save_pretrained_gguf("pog2_pedagogy_q4", tokenizer, quantization_method="q4_K_M")
```
