    intent: 'chroma' | 'eye' | 'generic';
    target?: string; // process name or file path

/**
 * VisionEngine Ã¢â‚¬â€ Multimodal observation substrate.
 * Implements SoverignEye (system-wide viewing) and Chromanumber (color/line mapping).
 */
export class VisionEngine extends EventEmitter {
    private static instance: VisionEngine;
    private readonly client: ModelClient;
    private readonly shell: ShellEngine;
    private readonly eye: SovereignEye;

    private static initialized = false;

    public static getInstance(config: SovereignConfig, renderer: Renderer): VisionEngine {
        if (!this.instance) {
            if (!config || !renderer) {
                throw new Error("VisionEngine requires config on first initialization");
            }
            this.instance = new VisionEngine(config, renderer);
            this.initialized = true;
        }
        return this.instance;
    }

    public static isInitialized(): boolean {
        return this.initialized;
    }

    private constructor(
        private readonly config: SovereignConfig,
        private readonly renderer: Renderer
    ) {
        super();
        this.client = new ModelClient(config);
        this.shell = new ShellEngine(renderer);
        this.eye = new SovereignEye('vision-engine');
        
        // Proxy SovereignEye captures to VisionEngine listeners (Switchboard)
        this.eye.on('capture', (res) => this.emit('capture', res));
    }

    async observe(params: VisionParams): Promise<Result<EngineOutput>> {
        let imagePath = '';

        // 1. Capture State (SoverignEye)
        if (params.intent === 'eye' || params.intent === 'chroma') {
            this.renderer.info('Capturing system state via SoverignEye...');
            const captureRes = await this.eye.capture({ type: 'window', target: params.target || 'POG2' });
            if (!captureRes.ok) return captureRes as any;
            if (captureRes.value && captureRes.value.imagePath) {
                imagePath = captureRes.value.imagePath;
            }
        }

        if (!imagePath) {
            imagePath = params.target && existsSync(params.target)
                ? params.target
                : join(tmpdir(), `POG2_obs_${Date.now()}.png`);
        }

        if (!existsSync(imagePath)) {
            return { ok: false, error: new Error(`Image not found for vision analysis: ${imagePath}`) };
        }

        // 2. Multimodal Analysis
        const base64Image = readFileSync(imagePath).toString('base64');
        const modelPrompt = this.buildVisionPrompt(params);

        const result = await this.client.call({
            model: params.modelOverride ?? this.config.visionModel,
            prompt: modelPrompt,
            images: [base64Image]
        });

        if (!result.ok) return result as any;

        return {
            ok: true,
            value: {
                output: result.value.response,
                data: {
                    intent: params.intent,
                    model: this.config.visionModel,
                    source: imagePath
                }
            }
        };
    }


    private buildVisionPrompt(params: VisionParams): string {
        if (params.intent === 'chroma') {
            return `
CHROMANUMBER TECHNICAL ANALYSIS:
You are the Chromanumber Substrate. Decompile this image into a rigorous technical color map.

TASKS:
1. COLOR KEY: Identify all unique HEX codes (#) used. Match them to UI tokens (Primary, Secondary, Background, Accent).
2. TOPOLOGY: Map the spatial coordinates of color regions (e.g., [0-100, 0-50] = #1A1A1A).
3. EDGE DETECTION: Identify the precise lines between contrasting colors.
4. KEY: Provide a final legend/key mapping colors to intended functions or elements.

OUTPUT FORMAT:
- HEX Palette List
- Region Grid Map
- Topology Summary
- Functional Key
`.trim();
        }

        if (params.intent === 'eye') {
            return `
You are SoverignEye. Analyze this system view/process screenshot.
TASKS:
1. Describe the active windows and visible content (HTML, HTTP, Localhost, Wrangler dev pages, etc.).
2. Detect any UI bugs, alignment issues, or console errors visible.
3. Provide a high-level summary of the current user workspace.
`.trim();
        }

        return params.prompt ?? "Analyze this image and describe its contents in technical detail.";
    }

    /**
     * Diagnostic probe for NodeTester.
     */
    public async healthCheck(): Promise<{ online: boolean; details: string; status: 'ONLINE' | 'DEGRADED' | 'OFFLINE' }> {
        const capture = await this.eye.capture({ type: 'window', target: 'POG2' });
        const online = capture.ok;
        return {
            online,
            status: online ? 'ONLINE' : 'DEGRADED',
            details: `Vision Substrate: ${online ? 'ONLINE' : 'OFFLINE'} (SoverignEye Win32 capture: ${online ? 'SUCCESS' : 'FAILED'})`
        };
    }
}

/**
 * SpatialVisionEngine Ã¢â‚¬â€ Enhanced vision with spatial mapping for Kimi
 * Provides line/color/space key maps so Kimi knows exactly what's on screen
 */
export class SpatialVisionEngine {
    private static instance: SpatialVisionEngine | undefined;
    private readonly client: ModelClient;
    private readonly shell: ShellEngine;
    private readonly eye: SovereignEye;
    private isObserving = false;


    private constructor(
        private readonly config: SovereignConfig,
        private readonly renderer: Renderer
    ) {
        if (SpatialVisionEngine.instance) {
            throw new Error('Singleton violation: use SpatialVisionEngine.getInstance()');
        }
        this.client = new ModelClient(config);
        this.shell = new ShellEngine(renderer);
        this.eye = new SovereignEye('spatial-vision');
    }

    private static initialized = false;

    public static getInstance(config?: SovereignConfig, renderer?: Renderer): SpatialVisionEngine {
        if (!this.instance) {
            if (!config || !renderer) {
                // Return a dormant instance shell to prevent boot crashes.
                // This allows the CNS to reference the engine before it is fully configured.
                return {
                    state: 'DORMANT',
                    healthCheck: () => ({ online: false, details: 'SpatialVisionEngine: DORMANT (Waiting for config/renderer)' }),
                    observe: async () => ({ ok: false, error: new Error('SpatialVisionEngine not initialized') }),
                    sampleFrame: async () => { },
                    verifySpatialAnchor: async () => ({ ok: false, error: new Error('SpatialVisionEngine not initialized') })
                } as any;
            }
            this.instance = new SpatialVisionEngine(config, renderer);
            this.initialized = true;
        }
        return this.instance;
    }

    public static isInitialized(): boolean {
        return this.initialized;
    }

    public static resetInstance(): void {
        if (process.env.NODE_ENV === 'production') {
            throw new Error('resetInstance is forbidden in production');
        }
        this.instance = undefined;
        this.initialized = false;
    }

    /**
     * sampleFrame - Perceptual reconciliation after a 600ms tick.
     */
    public async sampleFrame(): Promise<void> {
        logger.debug('Perceptual Frame Sampling started...');
        // Trigger UI analysis as the authoritative visual check
        await this.observe({
            intent: 'ui_analysis',
            includeScreenshot: true
        });
        logger.debug('Visual reconciliation complete. Pulse alignment confirmed.');
    }


    /**
     * verifySpatialAnchor - Visual ground-truth confirmation.
     * Ensures the 3D substrate has rendered the expected state after a pulse.
     */
    public async verifySpatialAnchor(): Promise<Result<boolean>> {
        const res = await this.observe({
            intent: 'ui_analysis',
            includeScreenshot: true
        });
        if (!res.ok) return err(res.error as Error);

        // In full impl, this would parse the spatialMap and check if the expected 
        // entity/coordinate markers are present.
        return ok(true);
    }

    async observe(params: SpatialVisionParams): Promise<Result<EngineOutput>> {
        if (this.isObserving) {
            return { ok: false, error: new Error('SpatialVisionEngine is already performing an observation. Pulse collision avoided.') };
        }
        this.isObserving = true;
        let imagePath = params.target && existsSync(params.target) ? params.target : '';
        try {
            // 1. Capture screen
            if (params.includeScreenshot !== false && !imagePath) {
                this.renderer.info('👁️   Spatial capture initiated...');

                // Phase 14: Dynamic port discovery via Switchboard singleton.
                const ports = [
                    SwitchboardLimb.getInstance().getActivePort() || 8788,
                    8787, // Wrangler Dev / Worker
                    8081  // rsmv Inspector Dev
                ];

                let lastErr: Error | null = null;
                for (const port of ports) {
                    const targets = [
                        `http://localhost:${port}/edge_radar.html`,
                        `http://localhost:${port}/globe`
                    ];

                    for (const targetUrl of targets) {
                        const captureRes = await this.eye.capture({ type: 'url', target: targetUrl });

                        if (captureRes.ok && captureRes.value.imagePath) {
                            imagePath = captureRes.value.imagePath;
                            break;
                        }
                        lastErr = captureRes.ok ? new Error(`Port ${port} path ${targetUrl} active but no screenshot produced`) : (captureRes.error as Error);
                    }
                    if (imagePath) break;
                }

                if (!imagePath) return { ok: false, error: lastErr || new Error('All vision dashboard ports (8788, 8787, 8081) failed to produce a capture.') };
            }

            if (!imagePath || !existsSync(imagePath)) {
                return { ok: false, error: new Error(`Image not found for spatial analysis: ${imagePath}`) };
            }

            // 2. Analyze based on intent
            const base64Image = readFileSync(imagePath).toString('base64');
            const modelPrompt = this.buildSpatialPrompt(params);

            const result = await this.client.call({
                model: this.config.visionModel,
                prompt: modelPrompt,
                images: [base64Image],
                yaoState: params.yaoState
            });

            if (!result.ok) {
                return { ok: false, error: result.error as any };
            }

            // 3. Parse spatial map if chroma/spatial intent
            let spatialMap: SpatialMap | undefined;
            if (params.intent === 'chroma' || params.intent === 'spatial_map' || params.intent === 'ui_analysis') {
                spatialMap = this.parseSpatialMap(result.value.response);
            }

            // 4. Chroma augmentation
            if (params.intent === 'chroma') {
                const deterministicColors = await this.getDeterministicColors(imagePath);
                if (deterministicColors.ok) {
                    spatialMap = spatialMap || { regions: [], lines: [], colorPalette: [] };
                    spatialMap.colorPalette = [
                        ...spatialMap.colorPalette,
                        ...deterministicColors.value.colors.map(c => ({
                            hex: c.hex,
                            usage: `Detected via pixel frequency analysis (freq: ${(c.frequency * 100).toFixed(1)}%)`,
                            locations: ['Global Substrate']
                        }))
                    ];
                }
            }

            // 5. Persist sensory probe metadata (Phase 112 Resilience)
            const captureId = uuidv4();
            SovereignKnowledgeDB.getInstance().recordCapture(
                captureId,
                Date.now(),
                'spatial_vision',
                imagePath,
                imagePath,
                { response: result.value.response, spatialMap },
                params
            );

            return {
                ok: true,
                value: {
                    output: result.value.response,
                    data: {
                        intent: params.intent,
                        spatialMap,
                        model: this.config.visionModel,
                        source: imagePath,
                        captureId
                    }
                }
            };
        } finally {
            if (imagePath && existsSync(imagePath) && !(params.target && params.target === imagePath)) {
                try {
                    unlinkSync(imagePath);
                    this.renderer.info(`🧹 Cleaned up temporary vision artifact: ${imagePath}`);
                } catch (e) {
                    this.renderer.warn(`Failed to clean up vision artifact: ${imagePath}`);
                }
            }
            this.isObserving = false;
        }
    }

    /**
     * Get dominant colors â€” delegates to ChromaCore.

     */
    private async getDeterministicColors(imagePath: string) {
        return ChromaCore.sample(imagePath, { gridSize: 50, maxColors: 10 });
    }

    private parseSpatialMap(response: string): SpatialMap | undefined {
        try {
            // Try to extract JSON from response
            const jsonMatch = response.match(/\{[\s\S]*\}/);
            if (jsonMatch) {
                const parsed = JSON.parse(jsonMatch[0]);
                return {
                    regions: parsed.regions || [],
                    lines: parsed.lines || [],
                    colorPalette: parsed.colorPalette || []
                };
            }
        } catch {
            // JSON parsing failed, return undefined
        }
        return undefined;
    }

    /**
     * Generate a text-based spatial map for Kimi to reference
     */
    generateTextMap(spatialMap: SpatialMap, width = 80, height = 24): string {
        const grid: string[][] = [];

        // Initialize empty grid
        for (let y = 0; y < height; y++) {
            grid[y] = new Array(width).fill(' ');
        }

        // Place regions
        for (const region of spatialMap.regions) {
            const rx = Math.floor((region.x / 1920) * width);
            const ry = Math.floor((region.y / 1080) * height);
            const rw = Math.max(1, Math.floor((region.width / 1920) * width));
            const rh = Math.max(1, Math.floor((region.height / 1080) * height));

            const char = region.type.charAt(0).toUpperCase();
            for (let y = ry; y < Math.min(ry + rh, height); y++) {
                for (let x = rx; x < Math.min(rx + rw, width); x++) {
                    if (y >= 0 && y < height && x >= 0 && x < width) {
                        grid[y]![x] = char;
                    }
                }
            }
        }

        // Draw lines
        for (const line of spatialMap.lines) {
            const sx = Math.floor((line.startX / 1920) * width);
            const sy = Math.floor((line.startY / 1080) * height);
            const ex = Math.floor((line.endX / 1920) * width);
            const ey = Math.floor((line.endY / 1080) * height);

            // Simple line drawing
            if (sx === ex) { // Vertical
                for (let y = Math.min(sy, ey); y <= Math.max(sy, ey) && y < height; y++) {
                    if (y >= 0 && sx >= 0 && sx < width) grid[y]![sx] = '|';
                }
            } else if (sy === ey) { // Horizontal
                for (let x = Math.min(sx, ex); x <= Math.max(sx, ex) && x < width; x++) {
                    if (sy >= 0 && sy < height && x >= 0) grid[sy]![x] = '-';
                }
            }
        }

        // Convert to string
        return grid.map(row => row.join('')).join('\n');
    }

    /**
     * Get summary of what's currently on screen for Kimi
     */
    async getScreenSummary(): Promise<Result<EngineOutput>> {
        return this.observe({
            intent: 'ui_analysis',
            includeScreenshot: true
        }) as any;
    }

        this.avatar = avatar;
    }

    /**
     * Diagnostic probe for NodeTester.

    public healthCheck(): { online: boolean; details: string; status: 'ONLINE' | 'DEGRADED' | 'OFFLINE' } {
        return {
            online: true,
            status: 'ONLINE',
            details: 'SpatialVisionEngine nominal; line/color decomposition substrate active.'
        };
    }

export class BlenderRenderer {
    private readonly rendersDir = join(getSovereignRoot(), 'renders');

    constructor(
        private readonly shell: ShellEngine,
        private readonly pulse: PulseMonitor,
        private readonly fallbackGen: (scene: StoryboardScene) => Promise<Result<{ path: string }>>
    ) {
        if (!existsSync(this.rendersDir)) {
            // mkdirSync(this.rendersDir, { recursive: true }); // Handled by shell if needed, but let's be safe
        }
    }

    /**
     * render - High-fidelity 3D synthesis with metabolic safety.
     */
    public async render(scene: StoryboardScene, projectPath: string): Promise<Result<Clip>> {
        const memory = await this.pulse.checkMemoryHealth();
        if (memory.success) {
            const usagePercent = Number((memory.data.total - memory.data.free) * 100n / memory.data.total);
            if (usagePercent > 92) {
                logger.warn({ usagePercent }, 'Metabolic Overload: >92% RAM. Triggering Ghost Fallback (2D).');
                return this.fallbackTo2D(scene);
            }
            logger.info({ usagePercent }, 'Metabolic headroom confirmed for 3D render.');
        }

        logger.info({ index: scene.index, description: scene.description }, 'Initiating Sovereign 3D Render.');

        const scriptPath = join(projectPath, `render_scene_${scene.index}.py`);
        const videoPath = join(projectPath, `scene_${scene.index}.mp4`);

        // Derive scene geometry and atmosphere from the storyboard description
        const sceneAnalysis = this.analyzeScene(scene);
        const pythonScript = this.generateCinemaScript(sceneAnalysis, videoPath);
        writeFileSync(scriptPath, pythonScript);
        logger.info({ scriptPath, bytes: pythonScript.length }, 'Cinema script written.');

        const blenderPath = `"C:\\\\Program Files\\\\Blender Foundation\\\\Blender 5.0\\\\blender.exe"`;
        const renderCmd = `${blenderPath} --background --python "${scriptPath}"`;

        const startTime = Date.now();
        const res = await this.shell.execute({
            command: renderCmd,
            cwd: projectPath,
            description: `3D Render Scene ${scene.index}`
        });

        if (!res.ok) {
            logger.error({ error: res.error.message }, 'Blender Render Failed. Falling back to 2D.');
            return this.fallbackTo2D(scene);
        }

        // Post-render verification: does the file actually exist?
        if (!existsSync(videoPath)) {
            logger.warn({ videoPath }, 'Blender exited clean but MP4 not found. Checking for frame output...');
            // Blender may write with suffix â€” check for any mp4 in the directory
            const { readdirSync } = await import('fs');
            const files = readdirSync(projectPath).filter(f => f.endsWith('.mp4'));
            if (files.length > 0) {
                const actualPath = join(projectPath, files[0]);
                logger.info({ actualPath }, 'Found rendered MP4 output.');
                const duration = (Date.now() - startTime) / 1000;
                return ok({ path: actualPath, duration, type: '3D_BLENDER' });
            }
            logger.error('No MP4 output found after render. Falling back to 2D.');
            return this.fallbackTo2D(scene);
        }

        const duration = (Date.now() - startTime) / 1000;
        return ok({ path: videoPath, duration, type: '3D_BLENDER' });
    }

    /**
     * analyzeScene - Extract rendering directives from scene description.
     */
    private analyzeScene(scene: StoryboardScene): SceneAnalysis {
        const combined = (scene.description + ' ' + (scene.visualPrompts?.[0] || '') + ' ' + (scene.cameraNotes || '')).toLowerCase();
        
        // Evolve: Detect injected geometry paths from ReimaginationEngine
        let injectedGeometry: string | null = null;
        const injectMatch = scene.description.match(/\[GEOMETRY_INJECT:(.*?)\]/);
        if (injectMatch) {
            injectedGeometry = injectMatch[1].trim();
        }

        const subjects: SubjectDirective[] = [];

        // Evolve: If geometry is injected, it becomes the primary subject.
        if (injectedGeometry) {
             subjects.push({ type: 'INJECTED', primitive: injectedGeometry });
        } else {
             // Retain original legacy generative parsing
             const subjectPatterns: [RegExp, string, string][] = [
                 [/dragon|kbd|wyrm|drake/i, 'DRAGON', 'mesh.primitive_cone_add'],
                 [/human|person|man|woman|knight|warrior|hero/i, 'HUMANOID', 'mesh.primitive_cylinder_add'],
                 [/tree|forest|wood/i, 'TREE', 'mesh.primitive_cone_add'],
                 [/castle|tower|building|dungeon/i, 'ARCHITECTURE', 'mesh.primitive_cube_add'],
                 [/sword|weapon|axe|blade/i, 'WEAPON', 'mesh.primitive_cylinder_add'],
                 [/fire|flame|lava|inferno/i, 'FIRE_EMITTER', 'mesh.primitive_ico_sphere_add'],
                 [/water|ocean|river|lake/i, 'WATER', 'mesh.primitive_plane_add'],

        // Derive color palette from description
        let keyColor: [number, number, number] = [0.8, 0.6, 0.3]; // warm default
        if (isFiery) keyColor = [1.0, 0.3, 0.05];
        else if (/ice|frost|cold|winter/i.test(combined)) keyColor = [0.2, 0.5, 1.0];

            isAerial,

    private async cleanupAuditOutput(maxFiles: number = 50): Promise<void> {
        try {
            const auditDir = join(process.cwd(), 'audit_output');
            if (!existsSync(auditDir)) return;

            const entries = await fsPromises.readdir(auditDir);
            const imageFiles = entries.filter(name => /\.(png|jpg|jpeg|webp|gif)$/i.test(name));

            if (imageFiles.length <= maxFiles) {
                return;
            }

            const imageStats = await Promise.all(imageFiles.map(async (name) => {
                const fullPath = join(auditDir, name);
                const stat = await fsPromises.stat(fullPath);
                return { name, path: fullPath, mtimeMs: stat.mtimeMs };
            }));

            imageStats.sort((a, b) => a.mtimeMs - b.mtimeMs);
            const deleteCount = imageStats.length - maxFiles;

            for (let i = 0; i < deleteCount; i += 1) {
                await fsPromises.unlink(imageStats[i].path);
            }

            logger.info({ deleted: deleteCount, location: 'audit_output', keep: maxFiles }, 'Chromatic engine pruned old audit artifacts');
        } catch (err) {
            logger.warn({ err }, 'Failed to prune audit_output artifacts');
        }
    }

    /**

     * @param maxColors - The maximum number of dominant colors to return.
     * @returns A Result containing an array of ColorSample objects.

    public async getDeterministicColors(imagePath: string, gridSize = 25, maxColors = 5): Promise<Result<readonly ColorSample[]>> {

    /**
     * Converts Hex string to RGB tuple.
     * @param hex - The hexadecimal color string (e.g., "#RRGGBB" or "RRGGBB").
     * @returns A readonly RGB tuple.
     */
    public static hexToRgb(hex: HexColor): RGB {
        const cleanHex = hex.replace('#', '');
        const bigint = parseInt(cleanHex, 16);
        return [
            (bigint >> 16) & 255,
            (bigint >> 8) & 255,
            bigint & 255
        ];
    }

    /**
     * Calculates Euclidean distance between two colors.
     * @param colorA - The first hexadecimal color string.
     * @param colorB - The second hexadecimal color string.
     * @returns The Euclidean distance between the two colors.
     */
    public static calculateDistance(colorA: HexColor, colorB: HexColor): number {
        const rgbA = this.hexToRgb(colorA);
        const rgbB = this.hexToRgb(colorB);

        return Math.sqrt(
            Math.pow(rgbA[0] - rgbB[0], 2) +
            Math.pow(rgbA[1] - rgbB[1], 2) +
            Math.pow(rgbA[2] - rgbB[2], 2)
        );
    }

    minX: number;
    maxX: number;
    minY: number;
    maxY: number;
    minZ: number;
    maxZ: number;

export interface MetricProfile {
    polyCount: number;
    vertexCount: number;
    aabb: AABB;
    dimensions: {
        width: number;
        height: number;
        depth: number;
    };
    volume: number;
    vertexDensity: number;
}

/**
 * SpatialMetricEngine â€” High-fidelity 3D geometric forensics
 * Calculates metrics directly from extracted ModelData buffers
 */
export class SpatialMetricEngine {

    /**
     * Calculates a comprehensive metric profile for a given 3D model
     */
    public static calculateProfile(model: ModelData): MetricProfile {
        let polyCount = 0;
        let vertexCount = 0;

        let minX = Infinity, maxX = -Infinity;
        let minY = Infinity, maxY = -Infinity;
        let minZ = Infinity, maxZ = -Infinity;

        for (const mesh of model.meshes) {
            // Polygon Count
            if (mesh.indices) {
                polyCount += mesh.indices.count / 3;
            }

            // Vertex Metrics & Bounding Box
            const pos = mesh.attributes.pos;
            if (pos) {
                vertexCount += pos.count;
                for (let i = 0; i < pos.count; i++) {
                    const x = pos.getX(i);
                    const y = pos.getY(i);
                    const z = pos.getZ(i);

                    if (x < minX) minX = x;
                    if (x > maxX) maxX = x;
                    if (y < minY) minY = y;
                    if (y > maxY) maxY = y;
                    if (z < minZ) minZ = z;
                    if (z > maxZ) maxZ = z;
                }
            }
        }

        // Handle empty models
        if (minX === Infinity) {
            minX = maxX = minY = maxY = minZ = maxZ = 0;
        }

        const width = maxX - minX;
        const height = maxY - minY;
        const depth = maxZ - minZ;
        const volume = width * height * depth;
        const vertexDensity = volume > 0 ? vertexCount / volume : 0;

        return {
            polyCount: Math.round(polyCount),
            vertexCount,
            aabb: { minX, maxX, minY, maxY, minZ, maxZ },
            dimensions: { width, height, depth },
            volume,
            vertexDensity
        };
    }

    /**
     * Diagnostic probe for system integrity
     */
    public healthCheck(): { online: boolean; details: string } {
        return {
            online: true,
            details: 'SpatialMetricEngine operational; geometric forensic substrate active.'
        };
    }
}

 * 2. Processes vision/intent via ChromanumberEngine.
 * 3. Actuates changes via ShaderInjector or RSMVController.

export class VisionActuationHarness {
    private eye: PuppeteerEye;
    private engine: ChromanumberEngine;
    private spatialVision: SpatialVisionEngine;
    private injector: ShaderInjector;
    private rsmv: RSMVController;
    private hudSubstrates: any[] = [];

    constructor(engine: ChromanumberEngine, spatialVision: SpatialVisionEngine) {
        this.engine = engine;
        this.spatialVision = spatialVision;
        this.eye = new PuppeteerEye();
        this.injector = new ShaderInjector();
        this.rsmv = RSMVController.getInstance();
        this.loadHUDSubstrates();
    }

    /**
     * Load pedagogical HUD candidates from the Sovereign drive.
     */
    private loadHUDSubstrates(): void {
        try {
            const pedagogyPath = getPedagogyPath();
            if (existsSync(pedagogyPath)) {
                const files = readdirSync(pedagogyPath);
                const hudFiles = files.filter(f => f.startsWith('GROUP_PRINCIPLE_HUD_CANDIDATE_'));

                for (const file of hudFiles) {
                    const data = JSON.parse(readFileSync(join(pedagogyPath, file), 'utf8'));
                    this.hudSubstrates.push(data);
                }
                logger.info({ count: this.hudSubstrates.length }, 'Loaded pedagogical HUD substrates.');
            }
        } catch (err) {
            logger.warn({ err }, 'Failed to load HUD substrates.');
        }
    }

    /**
     * Run a single perception-action cycle.
     */
    public async runCycle(url: string, prompt: string): Promise<Result<ActuationResult>> {
        logger.info({ url, prompt }, 'Starting Perception-Action Cycle');

        try {
            // 1. sensory Perception (DOM + Screenshot)
            const capture = await this.eye.domInspect(url);
            if (!capture.ok) return capture as any;

            logger.info({ screenshot: capture.value.screenshot }, 'Perception captured');

            // 1.5 Deterministic Color Detection (Phase 15.5)
            const colorsRes = await this.eye.getDeterministicColors(capture.value.screenshot);
            const dominantColors = colorsRes.ok ? colorsRes.value.map(c => c.hex) : [];
            if (dominantColors.length > 0) {
                logger.info({ dominantColors }, 'Deterministic colors detected from substrate');
            }

            // 2. Spatial UI Analysis (Phase 26)
            logger.info('Performing Spatial UI Analysis for HUD forensics');
            const spatialRes = await this.spatialVision.observe({
                intent: 'ui_analysis',
                target: capture.value.screenshot,
                includeScreenshot: false
            });

            if (spatialRes.ok && spatialRes.value.data?.spatialMap) {
                const map = spatialRes.value.data.spatialMap;
                logger.info({ regions: map.regions?.length }, 'Spatial HUD regions mapped');

                // Save mapping results to D:\ sovereign learning substrate
                const learningPath = resolveSovereignPath(join('learning', `hud_mapping_${Date.now()}.json`));
                writeFileSync(learningPath, JSON.stringify({
                    timestamp: Date.now(),
                    url,
                    spatialMap: map,
                    pedagogyMatches: this.hudSubstrates.length // Placeholder for match logic
                }, null, 2));
            }

            // 3. Intent Generation
            const forgeRes = await this.engine.forgeComponent(prompt, { spatialRes }, dominantColors);
            if (!forgeRes.ok) return forgeRes as any;

            const intent = forgeRes.value.actuation;
            if (!intent) {
                return { ok: true, value: { success: false, details: 'No actuation intent generated by engine.' } };
            }

            // 4. Actuation Execution
            logger.info({ action: intent.action }, 'Executing Actuation Intent');
            return await this.executeActuation(intent);

        } catch (err) {
            logger.error({ err }, 'Harness cycle failed');
            return { ok: false, error: err as Error };
        }
    }

    /**
     * Dispatch intent to the appropriate actuator.
     */
    private async executeActuation(intent: ActuationIntent): Promise<Result<ActuationResult>> {
        switch (intent.action) {
            case 'mutate_shader':
                return await this.injector.injectColor(intent.target, intent.parameters.code);

            case 'rsmv_control':
                return await this.rsmv.viewModel(intent.parameters.modelId);

            case 'actuate_hud':
                // High-fidelity HUD interaction (Mocked to click via Puppeteer in production)
                logger.info({ widgetId: intent.parameters.widgetId }, 'Actuating HUD Widget');
                return { ok: true, value: { success: true, details: `Actuated HUD Widget ${intent.parameters.widgetId}` } };

            default:
                return { ok: false, error: new Error(`Unsupported actuation action: ${intent.action}`) };
        }
    }

    /**
     * Graceful shutdown of sensory components.
     */
    public async shutdown(): Promise<void> {
        await this.eye.sleep();
        logger.info('Harness shutdown complete.');
    }
