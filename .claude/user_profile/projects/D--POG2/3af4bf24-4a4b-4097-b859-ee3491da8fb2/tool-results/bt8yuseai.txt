    private getSnapshotDir(): string {
        const dir = join(getSovereignRoot(), 'snapshots', 'edits');
        if (!existsSync(dir)) mkdirSync(dir, { recursive: true });
        return dir;
    }

    private snapshotYoung(filePath: string, content: string): string {
        const ts = Date.now();
        const name = basename(filePath).replace(/\.[^.]+$/, '');
        const snapPath = join(this.getSnapshotDir(), `${ts}_${name}.young.ts`);
        writeFileSync(snapPath, content, 'utf8');
        logger.info({ snapPath, filePath }, 'ðŸ“¸ Young state snapshot saved (before edit).');
        return snapPath;
    }

    private snapshotOld(filePath: string, content: string): string {
        const ts = Date.now();
        const name = basename(filePath).replace(/\.[^.]+$/, '');
        const snapPath = join(this.getSnapshotDir(), `${ts}_${name}.old.ts`);
        writeFileSync(snapPath, content, 'utf8');
        logger.info({ snapPath, filePath }, 'ðŸ“¸ Old state snapshot saved (after edit).');
        return snapPath;
    }

    /**
     * Hexagram Edit Gate: consults the current hexagram strategy.
     * Returns null if edit is permitted, or an error Result if blocked.
     */
    private checkHexagramGate(action: string, filePath: string): Result<EngineOutput> | null {
        const hexagram = this.router.getHexagram();
        const interpretation = hexagram.getInterpretation();
        const strategy = interpretation.strategy.split(':')[0]?.trim() || '';

        if (EDIT_BLOCKING_STRATEGIES.includes(strategy)) {
            logger.warn({ strategy, action, filePath }, 'ðŸš« Hexagram gate BLOCKED edit. Strategy forbids modification.');
            return {
                ok: false,
                error: new Error(`Hexagram Gate: ${strategy} strategy active (${interpretation.name}). Code modifications are suspended. Wait for state transition.`)
            };
        }
        return null;
    }

    /**
     * Phase 29: Safe write with Young/Old snapshots + function-deletion guard.
     * If the modified code deletes any exported symbol, the Young snapshot is restored.
     */
    private safeWriteWithGuard(filePath: string, originalContent: string, modifiedContent: string): Result<void> {
        // 1. Snapshot Young (before)
        const youngPath = this.snapshotYoung(filePath, originalContent);

        // 2. Function-Deletion Guard
        const archValidator = new ArchitecturalValidator(this.manifest);
        const preservation = archValidator.validatePreservation(originalContent, modifiedContent);
        if (!preservation.ok) {
            logger.error({ filePath, error: preservation.error?.reason }, 'ðŸ›¡ï¸ Function-Deletion Guard BLOCKED write. Restoring Young state.');
            // Restore original â€” the Young snapshot is the safety net
            writeFileSync(filePath, originalContent, 'utf8');
            return { ok: false, error: new Error(preservation.error?.reason || 'Function deletion detected') };
        }

        // 3. Write the modified content
        writeFileSync(filePath, modifiedContent, 'utf8');

        // 4. Snapshot Old (after)
        this.snapshotOld(filePath, modifiedContent);

        logger.info({ filePath, youngPath }, 'âœ… Safe write complete. Young/Old snapshots preserved.');
        return { ok: true, value: undefined };
    }

    async explain(params: ExplainParams): Promise<Result<EngineOutput>> {

${content}

\`\`\`
${content}

            model,

            return result as Result<EngineOutput>;

                if (!writeResult.ok) {
                    return { ok: false, error: writeResult.error! };

                // ARCHITECTURAL VALIDATION (post-write)
                const valid = await this.validation.validateAll(fixed, { fileName: params.filePath });

                    // Restore Young state on architectural violation
                    writeFileSync(params.filePath, content, 'utf8');
                    return { ok: false, error: new Error(`Fix interrupted by architectural violation: ${valid.error?.reason}. Young state restored.`) };
                }
            } catch (err) {
                return { ok: false, error: err as Error };
            }

            ok: true,
            value: {
                output: params.dryRun ? fixed : `Fixes applied to ${params.filePath}`,
                data: { model, dryRun: params.dryRun, yaoState: route.yaoState }
            }

    async smartEdit(params: SmartEditEngineParams): Promise<Result<EngineOutput>> {
        if (!existsSync(params.filePath)) return { ok: false, error: new Error(`File not found: ${params.filePath}`) };
        const content = readFileSync(params.filePath, 'utf8');

        // Phase 29: Hexagram Gate
        const gate = this.checkHexagramGate('smartEdit', params.filePath);
        if (gate) return gate;

        const result: SmartEditResult = await SmartEdit.calculateReplacement(content, {
            file_path: params.filePath,
            old_string: params.oldString,
            new_string: params.newString
        });

        if (result.occurrences === 0) {
            return {
                ok: false,
                error: new Error(`Could not find a match for the specified 'oldString' in ${params.filePath}. Use 'edit_code' for a full rewrite if indentation is too complex.`)
            };
        }

        const finalContent = SmartEdit.restoreTrailingNewline(content, result.newContent);

        try {
            // Phase 29: Safe write with Young/Old snapshots + function-deletion guard
            const writeResult = this.safeWriteWithGuard(params.filePath, content, finalContent);
            if (!writeResult.ok) {
                return { ok: false, error: writeResult.error! };
            }
            logger.info({ file: params.filePath, strategy: result.strategy }, 'SmartEdit applied with Young/Old snapshots.');
        } catch (err) {
            return { ok: false, error: err as Error };
        }

        return {
            ok: true,
            value: {
                output: `Successfully applied SmartEdit (${result.strategy} strategy) to ${params.filePath}. Found ${result.occurrences} occurrence(s).`,
                data: { strategy: result.strategy, occurrences: result.occurrences }
            }
        };

            details: `Analytic Substrate: Client(${clientOk}), Router(${routerOk})`
        };
    }
}

    async generateCode(params: GenerateCodeParams): Promise<Result<EngineOutput>> {
        const startTime = Date.now();

        // Existing file content injection if improving
        let existingContent = '';
        if (params.improve && params.targetFile && existsSync(params.targetFile)) {
            existingContent = readFileSync(params.targetFile, 'utf8');
        }

        const route = await this.router.route(params.complexity);
        const model = params.modelOverride ?? route.model;

        logger.info({ model, complexity: params.complexity, improve: params.improve }, 'Generating code');

        const prompt = params.improve && existingContent
            ? this.buildImprovePrompt(params.task, existingContent, params.language)
            : this.buildGeneratePrompt(params.task, params.language);

        // First pass
        const result = await this.client.call({
            model,
            prompt,
            systemPrompt: SOVEREIGN_SYSTEM_PROMPT,
            temperature: 0.2,
            category: 'orchestration'
        });

        if (!result.ok) {
            this.router.recordFailure(model);
            return result as Result<EngineOutput>;
        }
        this.router.recordSuccess(model);

        const generated = extractCodeBlock(result.value.response) ?? result.value.response;

        // Adversarial validation loop (Yang complexity or explicit flag)
        if (params.adversarial || params.complexity === 'Yang') {
            return this.adversarialLoop(generated, params.task, model, params.targetFile);
        }

        // Write to file if target specified
        if (params.targetFile) {
            try {
                writeFileSync(params.targetFile, generated, 'utf8');
                logger.info({ path: params.targetFile }, 'Code written to disk');

                // GROUND TRUTH VALIDATION
                const valid = await this.validation.validateAll(generated, { fileName: params.targetFile });
                if (!valid.ok) {
                    logger.warn({ error: valid.error?.reason }, 'Architectural validation REJECTED generated code');
                    return { ok: false, error: new Error(`Architectural Violation: ${valid.error?.reason}\nSuggestion: ${valid.error?.suggestion}`) };
                }
            } catch (err) {
                return { ok: false, error: err as Error };
            }
        }

        const latency = Date.now() - startTime;
        return {
            ok: true,
            value: {
                output: generated,
                data: {
                    model,
                    latency,
                    complexity: params.complexity,
                    writtenTo: params.targetFile ?? null,
                    yaoState: route.yaoState
                }
            }
        };
    }

    async generateDocs(params: GenerateDocsParams): Promise<Result<EngineOutput>> {

    private extractAdversarialScore(response: string): number {
        // 1. Primary Pattern: SCORE: N
        const primaryMatch = response.match(/SCORE:\s*\[?(\d+)\]?/i);
        if (primaryMatch?.[1]) return parseInt(primaryMatch[1]);

        // 2. Secondary Pattern: [N/100]
        const secondaryMatch = response.match(/\[(\d+)\/100\]/);
        if (secondaryMatch?.[1]) return parseInt(secondaryMatch[1]);

        // 3. Tertiary: Any lone digit >= 90 or specific success phrases
        if (response.toLowerCase().includes("it's clean, move out")) return 95;
        if (response.toLowerCase().includes("integrity verified")) return 90;

        // 4. Intelligent Fallback: 
        // If the model gave a long explanation but we can't find a score, 
        // give it a 75 (passing logic but not excellence) to prevent infinite loop.
        // If the response is very short, it's likely a failure (0).
        return response.length > 100 ? 75 : 0;
    }

    async atCursor(params: CursorParams): Promise<Result<EngineOutput>> {
        if (!existsSync(params.filePath)) return { ok: false, error: new Error(`File not found: ${params.filePath}`) };
        const content = readFileSync(params.filePath, 'utf8');
        const lines = content.split('\n');

        // Binary absolute: line must be within bounds
        if (params.line < 1 || params.line > lines.length) {
            return { ok: false, error: new Error(`Line ${params.line} out of range (file has ${lines.length} lines)`) };
        }

        // Build prefix/suffix context window (Ã‚Â±50 lines)
        const contextStart = Math.max(0, params.line - 51);
        const contextEnd = Math.min(lines.length, params.line + 49);
        const prefix = lines.slice(contextStart, params.line - 1).join('\n');
        const currentLine = lines[params.line - 1] ?? '';
        const suffix = lines.slice(params.line, contextEnd).join('\n');

        const cursorMark = currentLine.substring(0, params.column) + '<CURSOR>' + currentLine.substring(params.column);

        const route = await this.router.route('Yin');

- Each completion should be a self-contained, syntactically valid insertion at the cursor
- No placeholder text, no stubs

        const result = await this.client.call({
            model,
            prompt,
            systemPrompt: SOVEREIGN_SYSTEM_PROMPT,
            temperature: 0.2

            this.router.recordFailure(model);

    ) { }

    async execute(params: ShellParams): Promise<Result<EngineOutput>> {
        const cwd = params.cwd ?? process.cwd();
        const startMem = freemem();
        const startTime = performance.now();

        // Ternary Safety Gating
        if (this.router) {
            const hex = this.router.getHexagram();
            const contract = hex.getContract();
            const yao = hex.getYaoState();
            const directive = hex.getInterpretation().strategy;

            const destructivePatterns = /\b(rm\s+-[rf]|git\s+push\s+-f|del\s+\/q|rd\s+\/s|format\b|shred\b)/i;
            if (destructivePatterns.test(params.command)) {
                if (!contract.gates.allowDestructiveCommands) {
                    const msg = `â Œ COGNITIVE BLOCK: Destructive command rejected due to unstable/retreating state (Contract Restraint). Strategy: ${directive}`;
                    logger.warn({ command: params.command, yao, directive }, msg);
                    this.renderer.error(msg);
                    return { ok: false, error: new Error(msg) };
                }
            }

            if (yao === YaoState.OldYang) {
                logger.info({ command: params.command, directive }, 'PROJECTIVE EXECUTION: High-metabolism metabolic burst detected.');
            }
        }

        logger.info({ command: params.command, cwd }, 'Executing shell command');

        if (!params.silent) {
            this.renderer.info(`Executing: ${params.command}`);
        }

        try {
            // High-fidelity migration to spawn for reliable shell substrate access on Windows
            const shell = process.env.ComSpec || 'cmd.exe';
            const child = spawn(params.command, {
                cwd,
                env: { 
                    ...process.env, 
                    SystemRoot: process.env.SystemRoot || 'C:\\Windows',
                    SystemDrive: process.env.SystemDrive || 'C:',
                    POG2_ACTIVE: '1' 
                },
                shell: true
            });

            let stdout = '';
            let stderr = '';

            child.stdout?.on('data', (data: Buffer) => { stdout += data.toString(); });
            child.stderr?.on('data', (data: Buffer) => { stderr += data.toString(); });

            const exitCode = await new Promise<number | null>((resolve) => {
                child.on('close', (code) => resolve(code));
            });

            const endTime = performance.now();
            const duration = (endTime - startTime) / 1000;
            const endMem = freemem();
            const memDelta = Math.abs(startMem - endMem);

            if (exitCode !== 0) {
                logger.error({ error: stderr, exitCode, command: params.command }, 'Shell Error');
                return { 
                    ok: false, 
                    error: new Error(stderr || `Process exited with code ${exitCode}`)
                };
            }

            return {
                ok: true,
                value: {
                    output: stdout || '(empty output)',
                    data: {
                        stdout,
                        stderr,
                        exitCode: exitCode ?? 0,
                        memoryDelta: memDelta,
                        telemetry: {
                            durationMs: Math.round(duration * 1000),
                            memDeltaBytes: memDelta,
                            cpuLoad: loadavg()
                        }
                    }
                }
            };
        } catch (err: unknown) {
            const error = err as Error;
            const message = error.message || 'Unknown shell error';
            logger.error({ err, command: params.command }, 'Shell execution failed');

            return {
                ok: false,
                error: new Error(`Shell Error: ${message}`)
            };
        }
    }

 * Provides Kimi with direct access to edit, read, and navigate code
 * Integrated with TernaryRouter for metabolic context in the IDE.

export class VSCodeEngine {
    constructor(
        private readonly renderer: Renderer,
        private readonly router?: TernaryRouter
    ) { }

    async execute(params: VSCodeParams): Promise<Result<EngineOutput>> {
        switch (params.action) {
            case 'open':
                return this.openFile(params.filePath!);
            case 'read':
                return this.readFile(params.filePath!);
            case 'write':
                return this.writeFile(params.filePath!, params.content!, params.language);
            case 'edit':
                return this.editFile(params.filePath!, params.content!, params.line, params.column);
            case 'command':
                return this.runVSCodeCommand(params.command!);
            case 'getWorkspace':
                return this.getWorkspaceState();
            case 'setPosition':
                return this.setCursorPosition(params.filePath!, params.line!, params.column!);
            default:
                const action: string = params.action;
                return { ok: false, error: new Error(`Unknown VS Code action: ${action}`) };
        }
    }

    private async openFile(filePath: string): Promise<Result<EngineOutput>> {
        try {
            // Try to open in existing VS Code window
            await execAsync(`code --reuse-window "${filePath}"`);
            return {
                ok: true,
                value: {
                    output: `Opened ${filePath} in VS Code`,
                    data: { filePath, action: 'open' }
                }
            };
        } catch (error) {
            return { ok: false, error: new Error(`Failed to open file: ${error}`) };
        }
    }

    private async readFile(filePath: string): Promise<Result<EngineOutput>> {
        try {
            if (!existsSync(filePath)) {
                return { ok: false, error: new Error(`File not found: ${filePath}`) };
            }

            const content = readFileSync(filePath, 'utf8');
            const lines = content.split('\n');

            return {
                ok: true,
                value: {
                    output: content,
                    data: {
                        filePath,
                        lineCount: lines.length,
                        content
                    }
                }
            };
        } catch (error) {
            return { ok: false, error: new Error(`Failed to read file: ${error}`) };
        }
    }

    private async writeFile(
        filePath: string,
        content: string,
        language?: string
    ): Promise<Result<EngineOutput>> {
        try {
            writeFileSync(filePath, content, 'utf8');

            // Open the file after writing
            await this.openFile(filePath);

            return {
                ok: true,
                value: {
                    output: `Wrote ${content.length} chars to ${filePath}`,
                    data: { filePath, language, charCount: content.length }
                }
            };
        } catch (error) {
            return { ok: false, error: new Error(`Failed to write file: ${error}`) };
        }
    }

    private async editFile(
        filePath: string,
        newContent: string,
        line?: number,
        column?: number
    ): Promise<Result<EngineOutput>> {
        try {
            // If line/column specified, insert at position
            if (line !== undefined && existsSync(filePath)) {
                const existing = readFileSync(filePath, 'utf8');
                const lines = existing.split('\n');

                if (line < lines.length) {
                    const targetLine = lines[line] ?? '';
                    const col = column ?? targetLine.length;
                    lines[line] = targetLine.slice(0, col) + newContent + targetLine.slice(col);
                    writeFileSync(filePath, lines.join('\n'), 'utf8');
                } else {
                    // Append if beyond current lines
                    lines.push(newContent);
                    writeFileSync(filePath, lines.join('\n'), 'utf8');
                }
            } else {
                // Replace entire file
                writeFileSync(filePath, newContent, 'utf8');
            }

            return {
                ok: true,
                value: {
                    output: `Edited ${filePath}`,
                    data: { filePath, line, column }
                }
            };
        } catch (error) {
            return { ok: false, error: new Error(`Failed to edit file: ${error}`) };
        }
    }

    private async runVSCodeCommand(command: string): Promise<Result<EngineOutput>> {
        try {
            // Execute VS Code CLI command
            const { stdout, stderr } = await execAsync(`code ${command}`);

            return {
                ok: true,
                value: {
                    output: stdout || stderr || 'Command executed',
                    data: { command, stdout, stderr }
                }
            };
        } catch (error) {
            return { ok: false, error: new Error(`VS Code command failed: ${error}`) };
        }
    }

    private async getWorkspaceState(): Promise<Result<EngineOutput>> {
        try {
            // Try to get workspace info from current directory
            const { stdout: cwd } = await execAsync('cd');
            const currentDir = cwd.trim();

            // Check for .vscode folder or common project files
            const workspaceFolders = [currentDir];

            const state: WorkspaceState = {
                workspaceFolders,
                openFiles: [],
                activeEditor: undefined
            };

            // Try to detect active file from git or recent files
            try {
                const { stdout: gitFiles } = await execAsync('git diff --name-only 2>nul || echo ""');
                if (gitFiles.trim()) {
                    state.openFiles = gitFiles.split('\n').filter(f => f.trim());
                }
            } catch {
                // Git not available
            }

            return {
                ok: true,
                value: {
                    output: `Workspace: ${currentDir}\nFiles: ${state.openFiles.join(', ') || 'none detected'}`,
                    data: state as any
                }
            };
        } catch (error) {
            return { ok: false, error: new Error(`Failed to get workspace: ${error}`) };
        }
    }

    private async setCursorPosition(
        filePath: string,
        line: number,
        column: number
    ): Promise<Result<EngineOutput>> {
        try {
            // VS Code doesn't have a direct CLI for this, but we can open at line
            await execAsync(`code --goto "${filePath}:${line + 1}:${column + 1}"`);

            return {
                ok: true,
                value: {
                    output: `Positioned cursor at ${filePath}:${line + 1}:${column + 1}`,
                    data: { filePath, line, column }
                }
            };
        } catch (error) {
            return { ok: false, error: new Error(`Failed to set position: ${error}`) };
        }
    }

    /**
     * Get workspace files for Kimi to understand context
     */
    async getWorkspaceFiles(dir: string, depth = 2): Promise<string[]> {
        try {
            const { stdout } = await execAsync(
                `find "${dir}" -maxdepth ${depth} -type f \\( -name "*.ts" -o -name "*.js" -o -name "*.json" -o -name "*.md" \\) 2>nul || dir /s /b "${dir}\\*.ts" "${dir}\\*.js" "${dir}\\*.json" 2>nul`
            );
            return stdout.split('\n').filter(f => f.trim() && !f.includes('node_modules'));
        } catch {
            return [];
        }
    }

    public healthCheck(): { online: boolean; details: string } {
        return {
            online: true,
            details: 'VSCodeEngine active; ready for coordinated workspace operations.'
        };
    }

                     reasoning: `Observational Restraint: Critical structural break in [${filePath}], but system is in ${ternary} mode (Risk: ${riskTolerance}). Prioritizing manual developer observation.`
                 };
            }
            
            return {
                decision: CognitiveChoice.META,
                reasoning: `Critical Structural Break: ${error.message} in core manifest file [${filePath}]. Health: ${HealthStatus.Critical}.`
            };
        }

        // Cognitive Immunology Phase 3: Holistic Visual Diagnostic for UI
        const isUI = filePath?.toLowerCase().includes('renderer') || filePath?.toLowerCase().includes('ui') || filePath?.toLowerCase().includes('dashboard');
        if (isUI && ternary === CognitivePolarity.YANG && (this.visionEngine || this.spatialVisionEngine) && this.chromaEngine) {
             logger.info('UI fault detected during assertive Yang state. Invoking Visual/Spatial Sensory array.');
             await this.hexagramManager.pinCard(3, 'Visual Diagnostic', `Deploying Chromanumber and SpatialVision to physically inspect failure in ${filePath}`, yao);
             // We do not await full vision execution here, we just shape the decision heuristic to request physical inspection
             return {
                 decision: CognitiveChoice.EXTERNAL,
                 reasoning: `Somatic Diagnostic Required: Flaw detected in visual component [${filePath}]. Invoking SpatialVisionEngine & Chromanumber UI inspection before attempting an algorithmic patch.`
             };
        }

        await this.hexagramManager.pinCard(3, 'Neural Repair', `Healing ${filePath || 'system'}: ${senseCheck.analysis}`, YaoState.OldYang);

        return {
            decision: CognitiveChoice.SIMPLE,
            reasoning: `Correction Needed: ${senseCheck.analysis}. Status: ${BuildStatus.Warning}. Triggering auto-patch turn via ${this.correctionModel}.`
        };
    }

    private isFilePlanned(filePath: string, plan?: ExecutionPlan): boolean {
        if (!plan) return false;
        return plan.steps.some(step =>
            JSON.stringify(step.args).includes(basename(filePath)) ||
            (step.reasoning && step.reasoning.includes(basename(filePath)))
        );
    }

    private isModulePlanned(errorMessage: string, plan?: ExecutionPlan): boolean {
        if (!plan) return false;
        const match = errorMessage.match(/module '(.+?)'/);
        if (!match) return false;
        const moduleName = match[1]!;
        return plan.steps.some(step =>
            JSON.stringify(step.args).includes(moduleName) ||
            (step.reasoning && step.reasoning.includes(moduleName))
        );
    }

    private async performSenseCheck(error: TSCError): Promise<{ severity: string; analysis: string }> {
        const prompt = `Analyze this TSC error:
        Error: ${error.file}:${error.line} - [${error.code}] ${error.message}
        Determine if this is a minor type fix or a major structural break.
        Format: JSON { "severity": "low|medium|high|critical", "analysis": "concise explanation" }`;

        const result = await this.executor.call({ model: this.correctionModel, prompt });
        if (!result.ok) {
            return { severity: 'medium', analysis: 'Correction handler unavailable.' };
        }

        try {
            const jsonStr = result.value.response.match(/\{[\s\S]*\}/)?.[0] || '{}';
            const json = JSON.parse(jsonStr);
            return {
                severity: json.severity || 'medium',
                analysis: json.analysis || 'Generic type mismatch suspected.'
            };
        } catch {
            return { severity: 'medium', analysis: 'Failed to parse correction analysis.' };
        }
    }

    public async healWithProcess(projectRoot: string): Promise<{ healed: boolean; output: string }> {
        logger.info({ projectRoot }, 'Running biological verification (healWithProcess)');
        try {
            const { stdout, stderr } = await execAsync('npm run build', { cwd: projectRoot, timeout: 60000 });
            return { healed: true, output: stdout + stderr };
        } catch (error: unknown) {
            const err = error as { stdout?: string; stderr?: string };
            return { healed: false, output: (err.stdout || '') + (err.stderr || '') };
        }
    }

    /**
     * Phase 29: Hexagram-Driven Health Healing
     * Evaluates substrate degradation, checks the hexagram gate, and routes to CodeEngine if permissible.
     */
    public async healSubstrateError(nodeHealth: { path: string, role: string, category: string, status: string, details?: string }): Promise<{ decision: CognitiveChoice, reasoning: string }> {
        const hexagram = this.hexagramManager;
        const interpretation = hexagram.getInterpretation();
        const strategy = interpretation.strategy.split(':')[0]?.trim() || '';
        const yao = hexagram.getYaoState?.() || YaoState.YoungYang;

        logger.info({ node: nodeHealth.role, status: nodeHealth.status, strategy }, 'Attempting substrate healing...');

        const BLOCKING_STRATEGIES = ['FREEZE', 'ROLLBACK', 'PAUSE', 'DEEP_DORMANCY'];

        // 1. Hexagram Gate
        if (BLOCKING_STRATEGIES.includes(strategy) || yao === YaoState.OldMixed) {
            await hexagram.pinCard(2, 'Healing Suspended', `Node ${nodeHealth.role} (${nodeHealth.status}) auto-repair blocked by ${strategy} strategy / chaotic state.`, yao);
            return {
                decision: CognitiveChoice.META,
                reasoning: `Healing suspended due to ${strategy} strategy or OldMixed chaos. Node: ${nodeHealth.role} requires manual intervention.`
            };
        }

        // 2. Acknowledge intent to heal
        await hexagram.pinCard(2, 'Active Substrate Healing', `Initiating algorithmic repair protocols for ${nodeHealth.role}. Issue: ${nodeHealth.details || 'Unknown'}`, YaoState.OldYang);

        // 3. Provider/Endpoint Healing
        // For endpoints (e.g., 'ollama', 'cloudflare'), we route to the EndpointHealer
        if (nodeHealth.category === 'PROVIDER') {
            if ('getHealer' in this.executor) {
                const healer = (this.executor as any).getHealer();
                if (healer && typeof healer.diagnoseAndHeal === 'function') {
                    logger.info({ provider: nodeHealth.role }, 'Routing degraded provider to EndpointHealer');
                    const diag = await healer.diagnoseAndHeal(nodeHealth.role, nodeHealth.details || 'Unknown failure', undefined);
                    return {
                        decision: diag.resolved ? CognitiveChoice.SIMPLE : CognitiveChoice.META,
                        reasoning: diag.resolved 
                            ? `EndpointHealer successfully resolved ${nodeHealth.role}: ${diag.resolution}`
                            : `EndpointHealer could not resolve ${nodeHealth.role}: ${diag.resolution}`
                    };
                }
            }
        }

        // 4. Code-level Healing (Physical TS file fix)
        // Ensure path exists and is a .ts file (avoids trying to run CodeEngine on virtual nodes or directories)
        if (this.codeEngine && nodeHealth.path && nodeHealth.path.endsWith('.ts')) {
            const absolutePath = join(process.cwd(), nodeHealth.path);
            if (fs.existsSync(absolutePath)) {
                logger.info({ file: absolutePath }, 'Routing degraded node to CodeEngine.fix()');
                const fixRes = await this.codeEngine.fix({ filePath: absolutePath, dryRun: false });
                
                return {
                    decision: fixRes.ok ? CognitiveChoice.SIMPLE : CognitiveChoice.META,
                    reasoning: fixRes.ok ? `CodeEngine successfully applied algorithmic fix to ${nodeHealth.path}` : `CodeEngine algorithmic fix failed for ${nodeHealth.path}: ${fixRes.error}`
                };
            }
        }

        return {
             decision: CognitiveChoice.META,
             reasoning: `No explicit algorithmic repair path available for ${nodeHealth.role} (${nodeHealth.category}). Requires developer intervention.`
        };
    }

    public healthCheck(): { online: boolean; details: string } {
        const digOk = !!this.architectureDigest;
        const hexOk = !!this.hexagramManager;
        return {
            online: digOk && hexOk,
            details: `T-Cell Repair: Digest(${digOk}), Hexagram(${hexOk})`
        };
    }
