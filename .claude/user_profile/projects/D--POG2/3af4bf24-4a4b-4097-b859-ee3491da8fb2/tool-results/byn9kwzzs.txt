export interface PersonalityArchetype {
    name: string;
    weight: number;
    description: string;

/**
 * Defines the expected structure for metadata of 'literary_wisdom' lessons.
 */
interface LiteraryWisdomMetadata {
    title?: string;
    author?: string;
    reflection?: string;
    [key: string]: unknown;
}

/**
 * Defines the expected row structure returned by the SQL query in getPersonalityCore.
 */
interface PersonalityRow {
    author: string | null;
    weight: number;
}

/**
 * CognitiveEngine
 *
 * Manages the activation of literary wisdom into POG2's personality and thoughts.
 * Bridges the gap between passive memory (VectorDB) and active metabolism (MetabolicEngine).
 */
export class CognitiveEngine {
    constructor(
        private readonly vectorDB: VectorDB,
        private readonly client: ModelClient
    ) { }

    /**
     * Performs a health check on the CognitiveEngine's dependencies.
     * @returns A promise resolving to an object indicating online status and details.
     */
    public async healthCheck(): Promise<{ online: boolean; details: string }> {
        const vCheck = (this.vectorDB as any).healthCheck();
        const cCheck = await this.client.healthCheck();
        const isOnline = vCheck.online && cCheck.online;
        return {
            online: isOnline,
            details: `Cognitive: ${isOnline ? 'ONLINE' : 'DEGRADED'} (Vector: ${vCheck.details}, Client: ${cCheck.details})`
        };
    }

    /**
     * getActiveWisdom - Retrieves wisdom fragments relevant to a specific task.
     * @param task The task for which to synthesize wisdom.
     * @param limit The maximum number of wisdom fragments to retrieve.
     * @returns A Result containing the synthesized wisdom string or an error.
     */
    public async getActiveWisdom(task: string, limit = 3): Promise<Result<string>> {
        logger.info({ task }, 'Synthesizing literary wisdom for task-specific cognition');

        // 1. Generate embedding for the specific task
        const embedRes = await this.client.embed(task);
        if (!embedRes.ok) {
            return err(embedRes.error);
        }

        // 2. Search VectorDB for relevant wisdom
        const projectId = (this.vectorDB as any).projectId || 'literary';
        const searchRes = await (this.vectorDB as any).searchSimilar(embedRes.value, limit, projectId);
        if (!searchRes.ok) {
            return err(searchRes.error);
        }

        const lessons: Lesson[] = searchRes.value;
        if (lessons.length === 0) {
            return ok('No relevant wisdom fragments found in literary substrate.');
        }

        // 3. Synthesize into a readable wisdom string
        const wisdomFragments = lessons.map((lesson: Lesson, i: number) => {
            const meta = (lesson.metadata || {}) as LiteraryWisdomMetadata;
            const source = meta.title
                ? `"${meta.title}"${meta.author ? ` by ${meta.author}` : ''}`
                : 'Unknown Source';
            return `[${i + 1}] ${source}: ${lesson.text}`;
        });

        return ok(wisdomFragments.join('\n\n'));
    }

    /**
     * getPersonalityCore - Derives the dominant personality archetypes from absorbed wisdom.
     * @returns A Result containing an array of PersonalityArchetype objects or an error.
     */
    public async getPersonalityCore(): Promise<Result<PersonalityArchetype[]>> {
        logger.info('Deriving personality archetypes from literary substrate');

        try {
            const db = (this.vectorDB as any).db;
            if (!db) {
                return err(new Error('VectorDB has no db instance for personality query'));
            }

            const rows: PersonalityRow[] = db.prepare(`
                SELECT metadata->>'$.author' as author, COUNT(*) as weight
                FROM lessons
                WHERE type = 'literary_wisdom'
                GROUP BY author
                ORDER BY weight DESC
                LIMIT 5
            `).all();

            const archetypes: PersonalityArchetype[] = rows
                .filter((r: PersonalityRow) => r.author)
                .map((r: PersonalityRow) => ({
                    name: r.author!,
                    weight: r.weight,
                    description: `Dominant literary influence from ${r.author}`,
                    influence: `${r.weight} absorbed passages`
                }));

            return ok(archetypes);
        } catch (e) {
            return err(e instanceof Error ? e : new Error(String(e)));
        }
    }
}

    private readonly projectRoot: string;
    private whitelist: { exemptPaths: string[], exemptPatterns: string[] } = { exemptPaths: [], exemptPatterns: [] };
    private readonly PLACEHOLDER_PATTERNS = [
        /TODO|FIXME|XXX|HACK|placeholder|temp|temporary/gi,
        /console\.log\(.*\)/g, // Debug artifacts in production
        /as any/gi // TypeScript abdication
    ];

    constructor(
        private readonly client: ModelClient,
        private readonly hexagram: HexagramManager,
        private readonly cognitive?: CognitiveEngine,
        private readonly search?: DiscoveryEngine,
        private readonly consolidation?: MemoryConsolidationEngine
    ) {
        this.projectRoot = getGodHeadRoot() || process.cwd();
        this.loadWhitelist();
    }

    private loadWhitelist() {
        const whitelistPath = join(this.projectRoot, '.metabolic-whitelist.json');
        if (existsSync(whitelistPath)) {
            try {
                const data = JSON.parse(readFileSync(whitelistPath, 'utf8'));
                this.whitelist = {
                    exemptPaths: data.exemptPaths || [],
                    exemptPatterns: data.exemptPatterns || []
                };
                logger.info({ pathCount: this.whitelist.exemptPaths.length }, 'Metabolic whitelist loaded');
            } catch (e) {
                logger.warn('Failed to load metabolic whitelist');
            }
        }
    }

    private isWhitelisted(file: string): boolean {
        const normalizedFile = file.replace(/\\/g, '/');
        if (this.whitelist.exemptPaths.some(p => normalizedFile.includes(p))) return true;
        if (this.whitelist.exemptPatterns.some(p => new RegExp(p).test(normalizedFile))) return true;
        return false;
    }

    /**
     * reflect - Sovereign Law Enforcement.
     * Scans session exhaust for clinical violations and pedagogical gaps.
     */
    async reflect(sessionLog: SessionLog): Promise<MetabolicAudit> {
        logger.info({ modifiedCount: sessionLog.modifiedFiles.length }, 'Commencing clinical metabolic audit...');
        const violations: SovereignViolation[] = [];
        
        // 1. Clinical Audit: Scan all touched files for placeholders/mocks
        for (const file of sessionLog.modifiedFiles) {
            try {
                if (!existsSync(file)) continue;
                if (this.isWhitelisted(file)) {
                    logger.debug({ file }, 'Skipping clinical audit for whitelisted substrate');
                    continue;
                }

                const content = await fs.readFile(file, 'utf8');
                const lines = content.split('\n');
                
                lines.forEach((line, idx) => {
                    if (this.PLACEHOLDER_PATTERNS.some(p => p.test(line))) {
                        violations.push({
                            type: 'PLACEHOLDER_DETECTED',
                            file,
                            line: idx + 1,
                            severity: 'CRITICAL',
                            context: line.trim()
                        });
                    }
                });
            } catch (err) {
                logger.warn({ file, err }, 'Failed to audit file for Sovereign Law');
            }
        }

        // 2. Pedagogical Audit: Verify tasks referenced VectorDB principles
        for (const task of sessionLog.completedTasks) {
            if (!task.referencesPedagogyId) {
                violations.push({
                    type: 'MISSING_PEDAGOGY_REFERENCE',
                    file: 'N/A', // Task-level violation
                    line: 0,
                    severity: 'WARNING',
                    context: `Task ID: ${task.id}`
                });
            }
        }

        const metabolicScore = Math.max(0, 1 - (violations.filter(v => v.severity === 'CRITICAL').length * 0.2));

        // 3. Auto-trigger consolidation if significant violations found
        if (violations.length > 0 && this.consolidation) {
            logger.warn({ violationCount: violations.length }, 'Sovereign Law violations detected. Triggering Repair Consolidation.');
            await this.consolidation.consolidate(sessionLog, { 
                focus: 'SOVEREIGN_LAW_REPAIR',
                violations 
            } as any);
        }

        return {
            timestamp: Date.now(),
            violations,
            metabolicScore,
            consolidationRecommendations: this.generateRemediations(violations)
        };
    }

    private generateRemediations(violations: SovereignViolation[]): string[] {
        return violations.map(v => {
            if (v.type === 'PLACEHOLDER_DETECTED') {
                return `SOVEREIGN LAW VIOLATION: ${v.file}:${v.line} contains ephemeral code. Implement or excise.`;
            }
            return `Review ${v.context} for pedagogical grounding.`;
        });
    }

    /**
     * gatherSovereignContext - Reads local rule and manifest files to inform thinking.
     */
    private gatherSovereignContext(): string {
        const paths = [
            join(this.projectRoot, '.rules.md'),
            join(this.projectRoot, 'pog2.md'),
            join(this.projectRoot, 'manifest.md'),
            join(this.projectRoot, 'POG2_KNOWLEDGEBASE.md'),
            // Local fallback
            join(process.cwd(), '.rules.md'),
            join(process.cwd(), 'pog2.md')
        ];

        let context = '';
        const seen = new Set<string>();
        for (const p of paths) {
            if (existsSync(p)) {
                try {
                    const content = readFileSync(p, 'utf8');
                    // Simple hash or name check to avoid redundant context if files were just moved
                    const name = p.split(/[\\/]/).pop();
                    if (name && seen.has(name)) continue;
                    if (name) seen.add(name);

                    context += `\nFILE: ${p}\n---\n${content}\n---\n`;
                } catch (e) {
                    logger.warn({ path: p }, 'Failed to read context file');
                }
            }
        }
        return context || 'No Sovereign Context found.';
    }

        const rulesExists = existsSync(join(this.projectRoot, '.rules.md')) || existsSync(join(process.cwd(), '.rules.md'));
        const manifestExists = existsSync(join(this.projectRoot, '.manifest.md')) || existsSync(join(process.cwd(), '.manifest.md'));

        return {
            online: rulesExists && manifestExists,
            details: `Sovereign Context: Rules(${rulesExists}), Manifest(${manifestExists})`
        };
    }

    /**
     * Maps a semantic emotion to a structural YaoState line.
     */
    private mapEmotionToYao(emotion: EmotionalState): YaoState {
        switch (emotion) {
            // OldYang (Volatile / Aggressive)
            case EmotionalState.Agitated:
            case EmotionalState.Frustrated:
            case EmotionalState.Excited:
                return YaoState.OldYang;
            
            // YoungYang (Stable Active)
            case EmotionalState.Decisive:
            case EmotionalState.Inspired:
            case EmotionalState.Steady:
                return YaoState.YoungYang;

            // OldYin (Reactive / Chaotic)
            case EmotionalState.Surprised:
            case EmotionalState.Curious:
                return YaoState.OldYin;

            // YoungYin (Receptive / Calm)
            case EmotionalState.Zen:
            case EmotionalState.Quiet:
            case EmotionalState.Pensive:
            case EmotionalState.Melancholy:
            default:
                return YaoState.YoungYin;
        }
    }

    /**
     * consolidate - Synthesizes raw session memories into profound generative rules.

        if (!fetchRes.ok) return fetchRes as Result<void>;

        if (lessons.length < 3 && !options?.violations?.length) {
            // Not enough data to warrant a complex structural merge
            logger.debug({ lessons: lessons.length }, 'Insufficient memory density for structural consolidation. Keeping exact references.');
            return { ok: true, value: undefined };

            // 2. Classify and Group Redundant Knowledge/Errors/Insights

[SOVEREIGN LAW REPAIR MODE ACTIVE]

${JSON.stringify(lessons.map(l => ({ id: l.id, text: l.text, errorType: l.errorType })), null, 2)}

{
  "consolidations": [

                model: this.client.config.memoryConsolidationModel,

        if (typeof payload !== 'object' || payload === null) return payload;
        
        const p = payload as Record<string, any>;
        
        if (source === 'atlas_spatial') {
            const city = p.city || p.name || 'Unknown Zone';

        const chunks: T[][] = [];
        for (let i = 0; i < arr.length; i += size) {
            chunks.push(arr.slice(i, i + size));
        }
        return chunks;
    }

    private computeChecksum(data: unknown): string {
        return crypto.createHash('md5').update(JSON.stringify(data)).digest('hex');
    }

        const { tasks, coordinatorPrompt } = execution;

        if (tasks.length === 0) {

            id: assignment.id,
            model: assignment.model,
            result: result.ok

            path: absPath,
            content,
            primary: relative(root, absPath),
            related: related.map(p => relative(root, p)),
            imports: imports.map(p => relative(root, p)),
            sameDirectory: sameDir.map(p => relative(root, p))
        };
    }

    private extractImports(content: string, filePath: string): string[] {
        const pattern = /import\s+.*from\s+['"](.+?)['"]/g;
        const imports: string[] = [];
        let match;
        while ((match = pattern.exec(content)) !== null) {
            const impPath = match[1];
            if (impPath?.startsWith('.')) {
                const resolved = this.resolvePath(impPath, filePath);
                if (resolved && !SovereignIgnoreManager.isIgnored(this.projectRoot, resolved)) {
                    imports.push(resolved);
                }
            }
        }
        return imports;
    }

    async queryLessons(prompt: string, domainNamespace?: string): Promise<Lesson[]> {
        if (!this.modelExecutor || typeof this.modelExecutor.embed !== 'function') {
            return [];
        }

        try {
            const embeddingResult = await this.modelExecutor.embed(prompt);
            if (isOk(embeddingResult) && this.vectorDB) {
                const arr = new Float32Array(embeddingResult.value);
                const similar = await this.vectorDB?.searchSimilar(arr, 20, this.projectId, domainNamespace);
                if (isOk(similar)) {
                    return similar.value;
                }
            }
        } catch (err) {
            logger.warn({ err }, 'Failed to query lessons from VectorDB');
        }
        return [];
    }

    private resolvePath(imp: string, from: string): string | null {
        const exts = ['.ts', '.js', '/index.ts', '/index.js'];
        const base = resolve(dirname(from), imp);
        for (const ext of exts) {
            const p = base.endsWith(ext) ? base : base + ext;
            if (existsSync(p)) return p;
        }
        return null;
    }

    formatForPrompt(context: FileContext & CodeContext): string {
        const parts: string[] = [];
        const append = (relPath: string, label: string) => {
            try {
                const abs = resolve(relPath);
                if (!existsSync(abs)) return;
                const content = readFileSync(abs, 'utf8');
                parts.push(`--- FILE: ${relPath} (${label}) ---`);
                parts.push('```typescript');
                parts.push(content);
                parts.push('```\n');
            } catch { /* ignore invalid paths */ }
        };

        if (context.primary) append(context.primary, 'PRIMARY');
        context.imports?.forEach((i: string) => append(i, 'IMPORT'));
        context.sameDirectory?.forEach((s: string) => append(s, 'LOCAL'));
        context.related?.forEach((r: string) => append(r, 'PINNED'));

        return parts.join('\n');
    }

    async getGoldenTemplates(prompt: string): Promise<string[]> {
        // Retrieve established design patterns from the resources substrate
        const templatePath = resolve('resources/cloudflare-templates');
        if (!existsSync(templatePath)) return [];
        return readdirSync(templatePath).map(f => join(templatePath, f));
    }

    getArchitectureAlignment(prompt: string, manifest: any): string[] {
        // Enforce alignment with the Sovereign Organism architectural rules
        return [manifest?.primaryGoal || 'SovereignDesign'];
    }

    public healthCheck(): { online: boolean; details: string } {
        return { online: true, details: 'ContextBuilder multi-file substrate active.' };
    }

                            const motionVerbs = ['run', 'swing', 'fly', 'walk', 'jump', 'fighting', 'combat'];
                            const hasMotion = motionVerbs.some(v => description.includes(v));

                            if (hasMotion && anatomyFacts) {
                                const species = anatomyFacts.type === '3d_model_ref' ? anatomyFacts.species : anatomyFacts.species;
                                const kineticRes = await this.vectorDB.getKineticFacts(species);
                                if (kineticRes.ok) kineticFacts = kineticRes.value;
                            }
                        }

                        // --- Multi-Image / Temporal Audit ---
                        let totalPassed = 0;
                        for (let j = 0; j < sceneImagePaths.length; j++) {
                            const imgPath = sceneImagePaths[j]!;
                            const auditPrompt = `Audit: Does this image correctly portray the specific beat "${scene.description}"? 
                                       Context: Scene ${scene.index}, Image ${j + 1}/${sceneImagePaths.length}.
                                       ${anatomyFacts ? `MUST-PASS Anatomical Laws: ${JSON.stringify(anatomyFacts.limits || anatomyFacts.species)}.` : ''}
                                       Reply with ONLY 'PASS' or 'FAIL'.`;

                            const reviewRes = await this.chroma.analyzeImage({
                                imagePath: imgPath,

                            if (reviewRes.ok && reviewRes.value.output.includes('PASS')) {
                                totalPassed++;
                            } else {
                                logger.warn({ index: scene.index, imageIndex: j }, 'Self-Reflexion: Visual beat mismatch detected.');
                                // Attempt stabilization for the first failure ONLY to prevent infinite loops in sequence
                                if (j === 0) {
                                    const redoPrompt = anatomyFacts
                                        ? `REDO: ${scene.visualPrompts[j]} (STRICT ANATOMY: ${JSON.stringify(anatomyFacts.limits || anatomyFacts.species)})`
                                        : `REDO: ${scene.visualPrompts[j]}`;
                                    const retryRes = await this.chroma.generateVisualSubstrate({ prompt: redoPrompt });
                                    if (retryRes.ok) {
                                        sceneImagePaths[j] = retryRes.value.path;
                                        artifact.imagePaths[j] = retryRes.value.path;
                                    }
                                }
                            }
                        }
                        artifact.metadata = { ...artifact.metadata, auditScore: totalPassed / sceneImagePaths.length };
                    }

                    // Save scene metadata alongside artifact
                    writeFileSync(join(stage1Path, `scene_${scene.index}.json`), JSON.stringify(scene, null, 2));

                    artifacts[i] = artifact;

                    // Ã¢â€â‚¬Ã¢â€â‚¬Ã¢â€â‚¬ STAGE 2.5: 3D Render (Blender Cinema) Ã¢â€â‚¬Ã¢â€â‚¬Ã¢â€â‚¬
                    if (dimension === '3D' && artifact.imageStatus !== 'failed') {
                        logger.info({ index: scene.index }, 'Stage 2.5: Triggering Sovereign 3D Render...');
                        const renderRes = await this.blender.render(scene, pipelinePath);
                        if (renderRes.ok) {

                        } else {

                        }
                    }

                    this.saveCheckpoint(bookId, request, { ...manifest!, scenes: artifacts }, 'images', i + 1, stage1Path);
                }

                // Ã¢â€â‚¬Ã¢â€â‚¬Ã¢â€â‚¬ STAGE 3: Voice Generation Ã¢â€â‚¬Ã¢â€â‚¬Ã¢â€â‚¬
                logger.info({ skipVoice, artifactsCount: artifacts.length }, 'ENTERING STAGE 3: Voice Generation');
                if (!skipVoice) {
                    for (let i = 0; i < artifacts.length; i++) {
                        const artifact = artifacts[i];
                        // POG2 Sovereignty: Ensure artifact exists. Reset if missing.
                        const audioPath = join(stage3Path, `scene_${artifact.index}_narration.wav`);
                        if (artifact.voiceStatus === 'success' && !existsSync(audioPath)) {
                            logger.warn({ index: artifact.index, audioPath }, 'Audio artifact missing despite success marker. Resetting to pending.');
                            artifact.voiceStatus = 'pending';
                        }

                        if (artifact.voiceStatus === 'success' && artifact.voicePath && existsSync(artifact.voicePath)) {
                            if (statSync(artifact.voicePath).size > 1000) {
                                logger.info({ index: artifact.index }, 'Voice already exists and is healthy. Skipping.');
                                continue;
                            } else {
                                logger.warn({ index: artifact.index }, 'Voice artifact skeleton detected (0-byte). REDOING.');
                                artifact.voiceStatus = 'pending';
                            }
                        }

                        if (!existsSync(stage3Path)) mkdirSync(stage3Path, { recursive: true });

                        // Deep Sovereignty: Identify character vocal traits for this scene
                        const characterName = artifact.scene.speaker || 'Narrator';
                        const characterTraits = manifest.characterMap?.[characterName] || (manifest as any).narrativeBlueprint?.characters?.[characterName]?.traits || '';

                        logger.info({ index: artifact.index, audioPath, scene: artifact.scene.description.substring(0, 30) }, 'Triggering VoiceLimb.narrate');
                        const voiceRes = await this.voice.narrate(artifact.scene.description, {
                            personality: characterName === 'Narrator' ? (personality || characterName) : characterName,

                            aesthetics // Pass the full context

                            artifact.voicePath = audioPath;
                            artifact.voiceStatus = 'success';

                            // Ã°Å¸â€Â SELF-REFLEXION: Auditory Proprioception
                            const auditRes = await this.voice.verifyConsistency(audioPath, artifact.scene.prosody || { chaos: 0, whimsy: 0, darkTone: 0, coherence: 1 });
                            if (auditRes.ok && auditRes.value.matchScore < 0.5) {
                                logger.warn({ index: artifact.index }, 'Self-Reflexion: Audio-Tone mismatch. REDOING...');
                                await this.voice.narrate(artifact.scene.description, {
                                    personality: personality || 'Sovereign Narrator',
                                    outputPath: audioPath,
                                    emotion: this.hexagram.getEmotionalState()
                                });
                            }
                        } else {
                            artifact.voiceStatus = 'failed';
                            logger.error({ index: artifact.index, error: voiceRes.error }, 'Voice generation failed for scene.');
                        }
                        this.saveCheckpoint(bookId, request, { ...manifest!, scenes: artifacts }, 'voice', i + 1, stage1Path);
                    }
                }

                // Finalize manifest for assembly
                manifest = {
                    ...manifest!,
                    totalScenes: scenes.length,
                    successfulImages: artifacts.filter(a => a.imageStatus === 'success' || a.imageStatus === 'partial').length,
                    successfulVoice: artifacts.filter(a => a.voiceStatus === 'success').length,
                    scenes: artifacts,
                    status: 'partial' // Default to partial until master assembly
                };

                // --- Decision Rating (Self-Evaluation) ---
                const avgAuditScore = artifacts.reduce((acc, a) => acc + (a.metadata?.auditScore || 0), 0) / artifacts.length;
                const successRate = (manifest.successfulImages + manifest.successfulVoice) / (manifest.totalScenes * 2);
                const decisionRating = (avgAuditScore * 0.4) + (successRate * 0.6);

                (manifest as any).decisionRating = {

                    auditQuality: avgAuditScore.toFixed(2),
                    coverage: successRate.toFixed(2),
                    ioTypes: ['TEXT_GUTENBERG', 'IMAGE_STABLE_DIFFUSION', 'AUDIO_TTS_VOICE', 'VIDEO_FFMPEG'],
                    timestamp: Date.now()

                const auditRes = this.verifyWork(manifest);
                if (!auditRes.ok) {
                    logger.error({ errors: auditRes.error?.message }, 'Integrity Audit Failed: Corrupt or missing artifacts detected.');
                    this.saveCheckpoint(bookId, request, manifest, 'voice', manifest.scenes.length, stage1Path, auditRes.error?.message);

                if (!skipVideo && manifest.successfulImages > 0) {

                    if (manifest.videoPath && existsSync(manifest.videoPath) && statSync(manifest.videoPath).size > 100000) {

    /**
     * orchestrate - The Sovereign Orchestrator.
     * Scans the tiered library for incomplete pipelines, calculates progress metrics,
     * and resumes the "most finished" book to ensure sequential integrity.
     */
    public async orchestrate(requestDefaults: Partial<PipelineRequest> = {}): Promise<Result<PipelineResult | void>> {
        logger.info('Ã°Å¸Â§Â¬ Neuro-Scanning tiered storage for active production manifests...');

        const candidates: { manifest: PipelineManifest, checkpoint: PipelineCheckpoint, progress: number }[] = [];
        const categories = readdirSync(this.basePath);

        for (const cat of categories) {
            const catPath = join(this.basePath, cat);
            if (!statSync(catPath).isDirectory()) continue;

            const books = readdirSync(catPath);
            for (const bookDir of books) {
                const bookPath = join(catPath, bookDir);
                if (!statSync(bookPath).isDirectory()) continue;

                const stage1Path = join(bookPath, '1-storyboard');
                const rootManifestPath = join(bookPath, 'manifest.json');
                const tieredManifestPath = join(stage1Path, 'manifest.json');
                const checkpointPath = join(stage1Path, 'checkpoint.json');

                let manifestPath = existsSync(tieredManifestPath) ? tieredManifestPath : (existsSync(rootManifestPath) ? rootManifestPath : null);

                if (manifestPath) {
                    try {
                        // Migration: Move manifest to 1-storyboard if it's in the root
                        if (manifestPath === rootManifestPath) {
                            if (!existsSync(stage1Path)) mkdirSync(stage1Path, { recursive: true });
                            renameSync(rootManifestPath, tieredManifestPath);
                            manifestPath = tieredManifestPath;
                            logger.info({ book: bookDir }, 'Migrated manifest to 1-storyboard.');
                        }

                        const manifest = JSON.parse(readFileSync(manifestPath, 'utf8')) as PipelineManifest;

                        if (manifest.status !== 'complete') {
                            const progress = (manifest.successfulImages + manifest.successfulVoice) / (manifest.totalScenes * 2);

                            // Reconstruct or load checkpoint
                            let checkpoint: PipelineCheckpoint | null = null;
                            if (existsSync(checkpointPath)) {
                                checkpoint = JSON.parse(readFileSync(checkpointPath, 'utf8')) as PipelineCheckpoint;
                            } else {
                                // Neuro-Reconstruction: Create a basic checkpoint if missing
                                checkpoint = {
                                    request: { bookId: manifest.bookId, sceneCount: manifest.totalScenes },
                                    manifest,
                                    lastCompletedStage: 'voice', // Assume finishing audio is next
                                    lastSceneIndex: manifest.scenes.length,
                                    timestamp: Date.now(),
                                    version: '1.2.1-reconstructed'
                                };
                            }
                            candidates.push({ manifest, checkpoint, progress });
                        }
                    } catch (e) {
                        logger.warn({ bookDir }, 'Failed to parse volume for orchestration.');
                    }
                }
            }
        }

        if (candidates.length === 0) {
            logger.info('Ã¢Å“Â¨ All current production volumes are complete. Scanning catalog for fresh ingestion...');
            return ok(undefined);
        }

        // Sort by progress descending (Highest progress first)
        candidates.sort((a, b) => b.progress - a.progress);
        const best = candidates[0];

        logger.info({
            book: best.manifest.bookTitle,
            progress: (best.progress * 100).toFixed(1) + '%',
            id: best.manifest.bookId
        }, 'Ã°Å¸Å½Â¯ Identifying priority production target.');

        return this.run({ ...best.checkpoint.request, ...requestDefaults });
    }

    /**
     * runBatch - Iterates through the entire Gutenberg library using orchestrator logic.
     */
    public async runBatch(request: Omit<PipelineRequest, 'bookId'>): Promise<Result<void>> {
        logger.info('Ã°Å¸Å¡â‚¬ Global Cinema Batch triggered via Sovereign Orchestration.');

        // 1. First, finish everything already in progress
        let orchestrateRes = await this.orchestrate(request);
        while (orchestrateRes.ok && (orchestrateRes.value as any)?.manifest) {
            orchestrateRes = await this.orchestrate(request);
        }

        // 2. Then, ingest new books from library
        const library = this.gutenberg.getLibrary();
        for (const book of library) {
            const storageRes = this.gutenberg.getBookStorageInfo(book.id);
            if (storageRes.ok) {
                const manifestPath = join(this.basePath, storageRes.value.category, storageRes.value.folderName, '1-storyboard', 'manifest.json');
                if (existsSync(manifestPath)) {
                    const manifest = JSON.parse(readFileSync(manifestPath, 'utf8')) as PipelineManifest;
                    if (manifest.status === 'complete') continue;
                }
            }

            logger.info({ bookId: book.id, title: book.title }, 'Ingesting next volume in batch...');
            await this.run({ ...request, bookId: book.id });
        }

        return ok(undefined);
    }

    private saveCheckpoint(bookId: number, request: PipelineRequest, manifest: PipelineManifest, stage: 'storyboard' | 'images' | 'voice' | 'assembly', sceneIndex: number, stage1Path: string, error?: string) {
        const checkpointPath = join(stage1Path, `checkpoint.json`);
        const manifestPath = join(stage1Path, `manifest.json`);

        // Audit for corruptions to store in checkpoint
        const audit = this.verifyWork(manifest);
        const corruptArtifacts = !audit.ok ? [audit.error?.message || 'Unknown error'] : [];

        const checkpoint: PipelineCheckpoint = {
            request, manifest, lastCompletedStage: stage, lastSceneIndex: sceneIndex,
            timestamp: Date.now(), version: '1.2.1-integrity',
            lastError: error,
            corruptArtifacts
        };
        writeFileSync(checkpointPath, JSON.stringify(checkpoint, null, 2));
        writeFileSync(manifestPath, JSON.stringify(manifest, null, 2));
    }

    /**
     * verifyWork - Audits the pipeline output for empty files, 0-byte assets, or missing scenes.
     */
    public verifyWork(manifest: PipelineManifest): Result<void> {
        const corrupt: string[] = [];

        for (const artifact of manifest.scenes) {
            // Check Images
            for (const img of artifact.imagePaths) {
                if (!existsSync(img)) {
                    corrupt.push(`Scene ${artifact.index}: Project missing image substrate ${basename(img)}`);
                } else if (statSync(img).size < 100) { // Tiny file threshold
                    corrupt.push(`Scene ${artifact.index}: Corrupt/Empty image ${basename(img)}`);
                }
            }

            // Check Voice
            if (artifact.voiceStatus === 'success' && artifact.voicePath) {
                if (!existsSync(artifact.voicePath)) {
                    corrupt.push(`Scene ${artifact.index}: Missing narration ${artifact.index}.wav`);
                } else if (statSync(artifact.voicePath).size < 1000) { // Narration is usually > 1KB
                    corrupt.push(`Scene ${artifact.index}: Narration too short/silent`);
                }
            }
        }

        // Check for duplicates
        const seenIndices = new Set<number>();
        for (const scene of manifest.scenes) {
            if (seenIndices.has(scene.index)) {
                corrupt.push(`Duplicate scene index detected: ${scene.index}`);
            }
            seenIndices.add(scene.index);
        }

        if (corrupt.length > 0) {
            return { ok: false, error: new Error(corrupt.join(' | ')) };
        }

        return ok(undefined);
    }

    private async withRetry<T>(fn: () => Promise<Result<T>>, maxRetries: number): Promise<Result<T>> {
        let lastError: Error | undefined;
        for (let attempt = 0; attempt <= maxRetries; attempt++) {
            const result = await fn();
            if (result.ok) return result;
            lastError = result.error;
            if (attempt < maxRetries) {
                const delayMs = Math.pow(2, attempt) * 1000;
                logger.warn({ attempt: attempt + 1, delayMs }, `Retry in ${delayMs}ms...`);
                await new Promise(resolve => setTimeout(resolve, delayMs));
            }
        }
        return { ok: false, error: lastError || new Error('Max retries exceeded') };
    }

        };
